# [1차]: 이중 연결 리스트(Doubly Linked List) 모듈을 정의합니다.
# [2차]: 앞사람과 뒷사람의 손을 양쪽으로 붙잡고 늘어선 아이들의 기차 대형을 만듭니다.
"""
이중 연결 리스트 (Doubly Linked List) 모듈
dict, set, collections 없이 O(1) 삽입, 삭제, 이동 연산을 제공합니다.
"""

# [1차]: 개별 데이터와 양방향 링크(포인터)를 저장하는 Node 클래스를 선언합니다.
# [2차]: 기차 대형에서 앞뒤 친구의 손을 잡고 명찰을 달고 있는 개별 학생입니다.
class Node:
    """이중 연결 리스트 노드"""
    # [1차]: 노드 인스턴스를 초기화하며 key, value, data 및 이전/다음 포인터를 설정합니다.
    # [2차]: 학생이 가방에 물건(key, value)을 챙기고 앞뒤 손을 잡을 준비를 마칩니다.
    def __init__(self, key=None, value=None, data=None):
        # [1차]: 노드가 보유할 고유 식별자 key를 할당합니다.
        # [2차]: 사물함 번호나 이름표 명찰을 가슴에 답니다.
        self.key = key
        # [1차]: 노드가 보유할 데이터 값 value를 할당합니다.
        # [2차]: 이름표 뒤에 적힌 실제 연락처나 알맹이 정보입니다.
        self.value = value
        # [1차]: data 인자가 전달되면 우선 적용하고, 없으면 (key, value) 튜플로 기본 생성합니다.
        # [2차]: 종합 선물세트 상자에 키와 값을 한 번에 담아 보관합니다.
        self.data = data if data is not None else (key, value)
        # [1차]: 이전 노드를 가리키는 포인터 prev를 None으로 초기화합니다.
        # [2차]: 왼손으로 앞사람 옷자락을 잡기 전 손을 비워둡니다.
        self.prev = None
        # [1차]: 다음 노드를 가리키는 포인터 next를 None으로 초기화합니다.
        # [2차]: 오른손으로 뒷사람 옷자락을 잡기 전 손을 비워둡니다.
        self.next = None

    # [1차]: 객체의 문자열 표현형을 반환하여 디버깅을 용이하게 합니다.
    # [2차]: 학생의 이름표를 사람이 읽기 쉬운 글자로 보여줍니다.
    def __repr__(self):
        # [1차]: key와 value를 포맷팅한 문자열을 반환합니다.
        # [2차]: "저는 누구이고 어떤 값을 가지고 있어요"라고 자기소개합니다.
        return f"Node(key={self.key}, value={self.value})"


# [1차]: 센티넬 더미 노드를 기반으로 O(1) 연산을 보장하는 이중 연결 리스트 클래스를 선언합니다.
# [2차]: 출발선(Head)과 도착선(Tail) 깃발이 양 끝에 꽂혀 있는 안전한 달리기 트랙입니다.
class DoublyLinkedList:
    """
    더미 헤드(dummy head)와 더미 테일(dummy tail)을 사용하는 원자적 O(1) 이중 연결 리스트
    """
    # [1차]: DoublyLinkedList 인스턴스를 초기화하고 더미 헤드와 테일을 연결합니다.
    # [2차]: 트랙을 개장하고 출발선 기둥과 결승선 기둥을 줄로 단단히 이어둡니다.
    def __init__(self):
        # [1차]: 센티넬 더미 헤드 노드를 생성합니다.
        # [2차]: 달리기 트랙의 절대 움직이지 않는 출발 깃발 기둥입니다.
        self.head = Node()
        # [1차]: 센티넬 더미 테일 노드를 생성합니다.
        # [2차]: 달리기 트랙의 절대 움직이지 않는 도착 깃발 기둥입니다.
        self.tail = Node()
        # [1차]: 더미 헤드의 next 포인터가 더미 테일을 가리키도록 설정합니다.
        # [2차]: 출발 기둥의 밧줄을 도착 기둥으로 곧바로 연결합니다.
        self.head.next = self.tail
        # [1차]: 더미 테일의 prev 포인터가 더미 헤드를 가리키도록 설정합니다.
        # [2차]: 도착 기둥에서 되돌아보는 밧줄을 출발 기둥으로 맞물립니다.
        self.tail.prev = self.head
        # [1차]: 현재 저장된 실제 유효 노드의 개수를 나타내는 _size를 0으로 초기화합니다.
        # [2차]: 트랙 위를 달리는 실제 주자 수를 0명으로 기록합니다.
        self._size = 0

    # [1차]: 리스트의 맨 앞(head 바로 다음)에 노드를 O(1) 복잡도로 삽입합니다.
    # [2차]: 새 주자를 출발선 바로 맨 앞 1등 자리로 새치기시켜 합류시킵니다.
    def insert_front(self, node):
        # [1차]: 인자가 Node 객체가 아닌 경우 Node로 감싸서 인스턴스화합니다.
        # [2차]: 번호표를 안 단 주자에게 즉석에서 명찰과 번호표를 달아줍니다.
        if not isinstance(node, Node):
            # [1차]: 전달된 data를 기반으로 새 Node 인스턴스를 생성합니다.
            # [2차]: 새 선수의 옷에 등번호를 새겨 유니폼을 완성합니다.
            node = Node(data=node)

        # [1차]: 현재 더미 헤드의 바로 뒤에 위치한 첫 번째 유효 노드를 임시 참조합니다.
        # [2차]: 지금까지 출발선 바로 뒤에서 1등으로 서 있던 친구를 확인합니다.
        first = self.head.next
        # [1차]: 새 노드의 prev 포인터가 더미 헤드를 가리키게 합니다.
        # [2차]: 새 선수의 왼손으로 출발 기둥을 꼭 쥡니다.
        node.prev = self.head
        # [1차]: 새 노드의 next 포인터가 기존 첫 번째 노드를 가리키게 합니다.
        # [2차]: 새 선수의 오른손으로 기존 1등 친구의 손을 잡습니다.
        node.next = first
        # [1차]: 더미 헤드의 next 포인터가 새 노드를 가리키게 갱신합니다.
        # [2차]: 출발 기둥의 밧줄을 이제 새 선수에게 넘깁니다.
        self.head.next = node
        # [1차]: 기존 첫 번째 노드의 prev 포인터가 새 노드를 가리키게 갱신합니다.
        # [2차]: 기존 1등 친구의 왼손으로 새 선수의 손을 꼭 맞잡게 합니다.
        first.prev = node
        # [1차]: 저장된 유효 노드 개수를 1 증가시킵니다.
        # [2차]: 경기장의 출전 선수 명수 카운트를 1 올립니다.
        self._size += 1
        # [1차]: 삽입된 노드 객체를 반환합니다.
        # [2차]: 방금 합류한 선수를 가리켜 안내합니다.
        return node

    # [1차]: 리스트의 맨 뒤(tail 바로 이전)에 노드를 O(1) 복잡도로 삽입합니다.
    # [2차]: 새 주자를 결승선 바로 직전 맨 꼴찌 자리로 줄 서게 합니다.
    def insert_back(self, node):
        # [1차]: 인자가 Node 객체가 아니면 Node 인스턴스로 자동 래핑합니다.
        # [2차]: 명찰이 없는 사람에게 정식 선수 명찰을 달아줍니다.
        if not isinstance(node, Node):
            # [1차]: 전달된 데이터를 바탕으로 Node를 생성합니다.
            # [2차]: 번호표를 새로 발급합니다.
            node = Node(data=node)

        # [1차]: 현재 더미 테일의 바로 이전에 위치한 마지막 유효 노드를 임시 참조합니다.
        # [2차]: 지금까지 맨 꼴찌로 서 있던 친구를 확인합니다.
        last = self.tail.prev
        # [1차]: 새 노드의 prev 포인터가 기존 마지막 노드를 가리키게 합니다.
        # [2차]: 새 주자의 왼손으로 기존 꼴찌 친구의 손을 잡습니다.
        node.prev = last
        # [1차]: 새 노드의 next 포인터가 더미 테일을 가리키게 합니다.
        # [2차]: 새 주자의 오른손으로 결승선 기둥을 잡습니다.
        node.next = self.tail
        # [1차]: 기존 마지막 노드의 next 포인터가 새 노드를 가리키게 합니다.
        # [2차]: 기존 꼴찌 친구의 오른손을 새 주자에게 넘겨줍니다.
        last.next = node
        # [1차]: 더미 테일의 prev 포인터가 새 노드를 가리키게 갱신합니다.
        # [2차]: 결승선 기둥의 밧줄을 새 주자에게로 연결합니다.
        self.tail.prev = node
        # [1차]: 유효 노드 카운트를 1 증가시킵니다.
        # [2차]: 선수 명수를 1 올립니다.
        self._size += 1
        # [1차]: 삽입된 노드를 반환합니다.
        # [2차]: 방금 줄 선 주자를 반환합니다.
        return node

    # [1차]: 임의의 특정 노드를 O(1) 시간에 리스트의 연결 고리에서 제거합니다.
    # [2차]: 행렬 한가운데 있는 친구가 손을 놓고 조용히 대열에서 빠져나갑니다.
    def remove_node(self, node):
        # [1차]: 전달된 노드가 None이거나 포인터가 유효하지 않은 경우 무시하고 None을 반환합니다.
        # [2차]: 존재하지도 않는 허깨비를 빼내려고 하면 헛손질을 막습니다.
        if node is None or node.prev is None or node.next is None:
            # [1차]: 실패를 의미하는 None을 반환합니다.
            # [2차]: 아무 작업도 하지 않고 빈손으로 돌아옵니다.
            return None
        
        # [1차]: 센티넬 더미 노드(head, tail)는 절대 삭제되지 않도록 방어합니다.
        # [2차]: 출발 깃발 기둥이나 결승선 기둥을 뽑아가려는 시도를 철저히 차단합니다.
        if node is self.head or node is self.tail:
            # [1차]: None을 반환하여 보호합니다.
            # [2차]: 기둥 파손을 막고 종료합니다.
            return None

        # [1차]: 제거 대상 노드의 이전 노드를 변수에 보관합니다.
        # [2차]: 빠져나갈 친구의 앞사람을 확인합니다.
        prev_node = node.prev
        # [1차]: 제거 대상 노드의 다음 노드를 변수에 보관합니다.
        # [2차]: 빠져나갈 친구의 뒷사람을 확인합니다.
        next_node = node.next
        # [1차]: 이전 노드의 next 포인터가 다음 노드를 직접 가리키도록 건너뜁니다.
        # [2차]: 앞사람이 친구를 건너뛰어 뒷사람의 손을 직접 덥석 잡습니다.
        prev_node.next = next_node
        # [1차]: 다음 노드의 prev 포인터가 이전 노드를 직접 가리키도록 건너뜁니다.
        # [2차]: 뒷사람도 친구를 건너뛰어 앞사람의 손을 맞잡습니다.
        next_node.prev = prev_node

        # [1차]: 분리된 노드의 prev 링크를 None으로 해제합니다.
        # [2차]: 대열에서 나온 친구의 왼손을 깨끗이 씻습니다.
        node.prev = None
        # [1차]: 분리된 노드의 next 링크를 None으로 해제합니다.
        # [2차]: 대열에서 나온 친구의 오른손을 깨끗이 씻습니다.
        node.next = None
        # [1차]: 리스트의 크기를 1 감소시킵니다.
        # [2차]: 출전 주자 수 카운트를 1 차감합니다.
        self._size -= 1
        # [1차]: 분리된 노드 객체를 반환합니다.
        # [2차]: 대열을 떠난 주자를 밖으로 모셔옵니다.
        return node

    # [1차]: 맨 앞 유효 노드를 제거하고 반환합니다.
    # [2차]: 출발선 바로 뒤 1등 자리에 서 있던 주자를 밖으로 부릅니다.
    def remove_front(self):
        # [1차]: 리스트가 비어있는지 확인합니다.
        # [2차]: 트랙에 아무도 안 서 있는지 확인합니다.
        if self.is_empty():
            # [1차]: 비어있으면 None을 반환합니다.
            # [2차]: 데려올 사람이 없으니 빈손으로 돌아옵니다.
            return None
        # [1차]: 더미 헤드 바로 다음 노드를 remove_node로 삭제합니다.
        # [2차]: 1등 주자를 안전하게 대열에서 빼냅니다.
        return self.remove_node(self.head.next)

    # [1차]: 맨 뒤 유효 노드를 제거하고 반환합니다.
    # [2차]: 결승선 바로 앞 맨 꼴찌 자리에 서 있던 주자를 밖으로 부릅니다.
    def remove_back(self):
        # [1차]: 리스트가 비어있는지 검사합니다.
        # [2차]: 달리는 사람이 없는지 확인합니다.
        if self.is_empty():
            # [1차]: 비어있으면 None을 반환합니다.
            # [2차]: 뺄 사람이 없으니 빈손으로 돌아옵니다.
            return None
        # [1차]: 더미 테일 바로 이전 노드를 remove_node로 삭제합니다.
        # [2차]: 맨 꼴찌 주자를 안전하게 대열에서 뺍니다.
        return self.remove_node(self.tail.prev)

    # [1차]: 주어진 노드를 현재 위치에서 떼어내 맨 앞(head 바로 뒤)으로 O(1) 이동시킵니다.
    # [2차]: 달리기 중간에 있던 선수를 출발선 바로 뒤 최선두 자리로 순간이동시킵니다.
    def move_to_front(self, node):
        # [1차]: 노드가 없거나 센티넬 노드인 경우 연산을 중단합니다.
        # [2차]: 유령이거나 출발/결승선 기둥이면 이동시키지 않습니다.
        if node is None or node is self.head or node is self.tail:
            # [1차]: 함수를 즉시 종료합니다.
            # [2차]: 그대로 둡니다.
            return
        # [1차]: 이미 맨 앞에 있는 경우 불필요한 포인터 조작을 건너뜁니다.
        # [2차]: 이미 1등 자리에 서 있으면 굳이 움직이지 않습니다.
        if self.head.next is node:
            # [1차]: 조기 종료합니다.
            # [2차]: 제자리에 둡니다.
            return
        
        # [1차]: 노드의 원래 양옆 이웃 노드를 파악합니다.
        # [2차]: 이 선수가 서 있던 자리의 앞뒤 친구들을 봅니다.
        prev_node = node.prev
        # [1차]: 노드의 원래 다음 노드를 파악합니다.
        # [2차]: 뒷사람을 봅니다.
        next_node = node.next
        # [1차]: 원래 이웃 노드끼리 손을 잇습니다.
        # [2차]: 빠져나간 빈자리를 메우기 위해 앞뒤 친구가 손을 다시 잡습니다.
        prev_node.next = next_node
        # [1차]: 원래 이웃 노드끼리 역방향 손을 잇습니다.
        # [2차]: 뒷사람도 앞사람 손을 잡습니다.
        next_node.prev = prev_node

        # [1차]: 현재 1등 노드를 임시 저장합니다.
        # [2차]: 출발선 바로 뒤에 있던 현재 1등 친구를 확인합니다.
        first = self.head.next
        # [1차]: 노드의 prev를 더미 헤드로 연결합니다.
        # [2차]: 이동한 선수의 왼손으로 출발 깃발을 잡습니다.
        node.prev = self.head
        # [1차]: 노드의 next를 기존 1등 노드로 연결합니다.
        # [2차]: 이동한 선수의 오른손으로 기존 1등 친구의 손을 잡습니다.
        node.next = first
        # [1차]: 더미 헤드의 next를 해당 노드로 갱신합니다.
        # [2차]: 출발 깃발의 밧줄을 이제 이 선수에게 연결합니다.
        self.head.next = node
        # [1차]: 기존 1등 노드의 prev를 해당 노드로 갱신합니다.
        # [2차]: 기존 1등 친구가 새 선수의 손을 잡습니다.
        first.prev = node

    # [1차]: 리스트의 총 노드 수를 반환합니다.
    # [2차]: 트랙 위에서 달리고 있는 전체 선수의 머릿수를 알려줍니다.
    def size(self):
        # [1차]: 내부 크기 변수 _size를 반환합니다.
        # [2차]: 기록해둔 주자 숫자를 말해줍니다.
        return self._size

    # [1차]: 리스트가 비어있는지 여부를 불리언 값으로 반환합니다.
    # [2차]: 운동장에 주자가 0명인지 묻는 질문에 참/거짓으로 답합니다.
    def is_empty(self):
        # [1차]: _size가 0인지 비교합니다.
        # [2차]: 텅 비었는지 확인합니다.
        return self._size == 0

    # [1차]: 리스트의 노드들을 순회하여 파이썬 기본 리스트로 변환합니다.
    # [2차]: 줄 서 있는 모든 선수들의 명단을 종이에 순서대로 적어 뽑아냅니다.
    def to_list(self):
        # [1차]: 결과를 담을 빈 리스트를 초기화합니다.
        # [2차]: 깨끗한 백지 명부를 준비합니다.
        result = []
        # [1차]: 첫 번째 유효 노드부터 순회를 시작합니다.
        # [2차]: 1등 주자부터 시작합니다.
        curr = self.head.next
        # [1차]: 더미 테일에 도달할 때까지 포인터를 전진합니다.
        # [2차]: 결승선 기둥에 닿을 때까지 다음 사람으로 걸어갑니다.
        while curr is not self.tail:
            # [1차]: 현재 노드를 결과 리스트에 추가합니다.
            # [2차]: 명부에 주자 이름을 한 줄 적습니다.
            result.append(curr)
            # [1차]: 다음 노드로 이동합니다.
            # [2차]: 뒷사람에게로 발걸음을 옮깁니다.
            curr = curr.next
        # [1차]: 수집된 노드 리스트를 반환합니다.
        # [2차]: 완성된 명부를 건넵니다.
        return result

    # [1차]: 파이썬의 for ... in 문에서 직접 순회할 수 있도록 이터레이터를 구현합니다.
    # [2차]: 손님들이 줄 선 선수들을 앞에서부터 한 명씩 차례대로 만나볼 수 있게 합니다.
    def __iter__(self):
        # [1차]: 첫 번째 유효 노드를 시작점으로 잡습니다.
        # [2차]: 맨 앞 1등 주자부터 섭외합니다.
        curr = self.head.next
        # [1차]: 더미 테일 이전까지 반복합니다.
        # [2차]: 결승선 직전까지 한 명씩 넘깁니다.
        while curr is not self.tail:
            # [1차]: 현재 노드를 yield로 방출합니다.
            # [2차]: 이번 주자를 차례대로 호명합니다.
            yield curr
            # [1차]: 다음 노드로 포인터를 이동합니다.
            # [2차]: 다음 주자를 찾아갑니다.
            curr = curr.next

# [1차]: 체이닝 기반 해시맵 모듈을 정의합니다.
# [2차]: 여러 개의 번호표 우편함과 서류철을 이용해 물건을 광속으로 찾아주는 수납장을 만듭니다.
"""
체이닝 기반 해시맵 (Chaining Hash Map) 모듈
dict, set, collections 없이 DoublyLinkedList를 버킷 체이닝으로 사용하여 구현합니다.
로드 팩터 0.75 초과 시 버킷을 2배 확장(Rehashing)합니다.
"""

# [1차]: 체이닝 버킷 구현에 필요한 DoublyLinkedList 및 Node 클래스를 임포트합니다.
# [2차]: 우편함 서류철에 서류를 엮어둘 이중 기차 대형 부품들을 가져옵니다.
from src.doubly_linked_list import DoublyLinkedList, Node


# [1차]: 고유 해시 함수와 체이닝, 동적 리사이징을 갖춘 HashMap 클래스를 선언합니다.
# [2차]: 내장 딕셔너리 없이 밑바닥부터 세운 대형 우편 관리소입니다.
class HashMap:
    """
    체이닝(Chaining) 방식 해시맵
    - 충돌 해결: DoublyLinkedList 체이닝
    - 해시 함수: djb2 기반 고분산 해시 알고리즘
    - 동적 리사이징: 로드 팩터 > 0.75 시 2배 확장
    """
    # [1차]: 기본 버킷 크기 상수를 16으로 정의합니다.
    # [2차]: 최초에 열어두는 우편함 개수를 16칸으로 맞춥니다.
    DEFAULT_CAPACITY = 16
    # [1차]: 버킷 리사이징 임계값인 최대 로드 팩터 상수를 0.75로 정의합니다.
    # [2차]: 우편함 공간이 75% 이상 차면 칸을 늘리겠다는 안전 기준선입니다.
    MAX_LOAD_FACTOR = 0.75

    # [1차]: HashMap 객체를 초기화하며 버킷 배열을 설정합니다.
    # [2차]: 우편함들을 일렬로 세우고 장부를 0으로 초기화합니다.
    def __init__(self, initial_capacity=DEFAULT_CAPACITY):
        # [1차]: 초기 수용량(버킷 수)을 최소 4 이상으로 안전하게 보정합니다.
        # [2차]: 적어도 4칸 이상의 우편함 칸수를 보장합니다.
        self._capacity = max(initial_capacity, 4)
        # [1차]: 저장된 총 고유 키의 개수를 0으로 초기화합니다.
        # [2차]: 보관된 우편물 개수를 0개로 설정합니다.
        self._size = 0
        # [1차]: 각 버킷 인덱스마다 빈 DoublyLinkedList를 할당한 리스트를 생성합니다.
        # [2차]: 16개의 우편함 칸마다 각각 전용 서류철(연결 리스트)을 꽂아둡니다.
        self._buckets = [DoublyLinkedList() for _ in range(self._capacity)]

    # [1차]: djb2 해시 알고리즘을 사용하여 임의의 키를 32비트 양의 정수 해시코드로 변환합니다.
    # [2차]: 주민등록번호나 이름을 소수 5381과 마법의 곱셈 공식에 넣어 고유 번호표로 바꿉니다.
    def _hash(self, key):
        """
        직접 설계한 djb2 기반 해시 함수
        소수 5381과 비트 시프트 연산(h * 33 + ord(c))을 통해 균등한 해시 분포를 보장합니다.
        """
        # [1차]: 키를 문자열로 정규화합니다.
        # [2차]: 어떤 종류의 이름표든 글자로 또박또박 바꿉니다.
        key_str = str(key)
        # [1차]: djb2 알고리즘의 초기 소수 시드값 5381을 설정합니다.
        # [2차]: 마법의 계산기 기본 밑자리 숫자를 5381로 잡습니다.
        h = 5381
        # [1차]: 문자열의 각 문자를 순회합니다.
        # [2차]: 이름표의 글자를 한 글자씩 읽어갑니다.
        for char in key_str:
            # [1차]: (h * 33 + ord(char)) 연산을 비트 시프트로 수행하고 31비트 양수로 마스킹합니다.
            # [2차]: 숫자를 마구 흔들어 섞은 뒤 음수가 되지 않도록 깨끗이 정돈합니다.
            h = (((h << 5) + h) + ord(char)) & 0x7FFFFFFF
        # [1차]: 최종 계산된 해시값을 반환합니다.
        # [2차]: 골고루 흩어진 고유 번호표를 줍니다.
        return h

    # [1차]: 해시값을 현재 버킷 크기로 모듈러 연산하여 0 ~ (capacity - 1) 범위의 인덱스를 산출합니다.
    # [2차]: 번호표를 우편함 개수로 나누어 몇 번 칸에 넣을지 결정합니다.
    def _get_bucket_index(self, key):
        """해시값을 현재 버킷 크기로 모듈러 연산하여 인덱스 계산"""
        # [1차]: _hash(key) % self._capacity 연산 결과를 반환합니다.
        # [2차]: "당신 서류는 3번 우편함에 넣으세요" 하고 칸 번호를 알려줍니다.
        return self._hash(key) % self._capacity

    # [1차]: 키-값 쌍을 저장하며 기존 키는 덮어쓰고, 로드팩터 초과 시 2배 리사이징을 수행합니다.
    # [2차]: 우편함에 서류를 꽂고, 꽉 차기 직전이면 수납장을 2배 큰 새 수납장으로 교체합니다.
    def put(self, key, value):
        """
        키와 값을 해시맵에 저장합니다.
        이미 키가 존재하면 값을 덮어씁니다.
        저장 후 로드 팩터가 0.75를 초과하면 버킷을 2배 확장합니다.
        """
        # [1차]: 키가 들어갈 버킷 인덱스를 계산합니다.
        # [2차]: 몇 번 우편함으로 가야 할지 찾습니다.
        idx = self._get_bucket_index(key)
        # [1차]: 해당 인덱스의 DoublyLinkedList 버킷을 가져옵니다.
        # [2차]: 해당 칸의 서류철을 꺼냅니다.
        bucket = self._buckets[idx]

        # [1차]: 해당 버킷에 이미 동일한 키를 가진 노드가 존재하는지 선형 순회합니다.
        # [2차]: 서류철을 넘기며 이미 같은 이름표가 꽂혀 있는지 살펴봅니다.
        for node in bucket:
            # [1차]: 키가 일치하는 노드를 발견한 경우 분기합니다.
            # [2차]: 예전에 보관한 같은 이름표 서류를 찾았습니다.
            if node.key == key:
                # [1차]: 덮어쓰기 전 기존 value를 임시 저장합니다.
                # [2차]: 옛날 서류 알맹이를 잠시 메모해둡니다.
                old_val = node.value
                # [1차]: 노드의 value를 새 값으로 갱신합니다.
                # [2차]: 새 내용물로 종이를 싹 갈아 끼웁니다.
                node.value = value
                # [1차]: 노드의 data 튜플도 새 값으로 갱신합니다.
                # [2차]: 종합 선물세트 상자의 내용도 업데이트합니다.
                node.data = (key, value)
                # [1차]: 기존 값을 반환하며 종료합니다.
                # [2차]: 교체된 옛날 물건을 건네주고 끝냅니다.
                return old_val

        # [1차]: 버킷에 키가 없으므로 새 Node 객체를 생성합니다.
        # [2차]: 신규 회원의 새 서류 봉투를 만듭니다.
        new_node = Node(key=key, value=value, data=(key, value))
        # [1차]: 버킷 리스트의 맨 뒤에 새 노드를 체이닝 삽입합니다.
        # [2차]: 서류철 맨 뒷장에 새 서류 봉투를 착 꽂아 넣습니다.
        bucket.insert_back(new_node)
        # [1차]: 전체 키 개수를 1 증가시킵니다.
        # [2차]: 총 보관 서류 개수 장부를 1 올립니다.
        self._size += 1

        # [1차]: 현재 로드 팩터(size / capacity)가 0.75를 초과하는지 검사합니다.
        # [2차]: 수납장에 서류가 75% 넘게 차서 비좁아졌는지 확인합니다.
        if (self._size / self._capacity) > self.MAX_LOAD_FACTOR:
            # [1차]: 버킷 용량을 2배로 확장하고 재해싱합니다.
            # [2차]: 우편함을 2배로 넓히고 서류들을 널찍하게 다시 정리합니다.
            self._resize(self._capacity * 2)

        # [1차]: 신규 키 삽입이므로 이전 값 없음을 뜻하는 None을 반환합니다.
        # [2차]: 새로 들어온 손님이니 옛날 물건은 없다고 알립니다.
        return None

    # [1차]: 지정된 키에 연관된 값을 O(1) 평균 시간으로 조회합니다.
    # [2차]: 이름표를 보고 우편함으로 달려가 서류 내용을 꺼내 읽어줍니다.
    def get(self, key):
        """
        키에 해당하는 값을 조회합니다.
        키가 존재하지 않으면 None을 반환합니다.
        """
        # [1차]: 키의 버킷 인덱스를 계산합니다.
        # [2차]: 몇 번 우편함 칸에 있는지 번호를 찍습니다.
        idx = self._get_bucket_index(key)
        # [1차]: 해당 버킷의 연결 리스트를 가져옵니다.
        # [2차]: 그 칸의 서류철을 꺼냅니다.
        bucket = self._buckets[idx]

        # [1차]: 버킷 내 노드들을 순회하며 key가 일치하는 노드를 찾습니다.
        # [2차]: 서류철을 넘기며 일치하는 이름표를 찾습니다.
        for node in bucket:
            # [1차]: 일치하는 키를 찾으면 해당 노드의 value를 반환합니다.
            # [2차]: 찾았으면 서류 안의 내용을 읽어드립니다.
            if node.key == key:
                # [1차]: 노드의 value를 반환합니다.
                # [2차]: 알맹이를 꺼내 드립니다.
                return node.value
        # [1차]: 일치하는 키가 없으면 None을 반환합니다.
        # [2차]: 없으면 "그런 손님 서류는 없습니다" 하고 빈손으로 돌아옵니다.
        return None

    # [1차]: 키에 해당하는 Node 객체 레퍼런스 자체를 반환합니다 (LRU 연동용).
    # [2차]: 내용물뿐만 아니라 서류 봉투 자체를 통째로 가리켜줍니다.
    def get_node(self, key):
        """
        키에 해당하는 노드 참조 자체를 조회합니다 (LRU 등 내부 연동용).
        """
        # [1차]: 버킷 인덱스를 계산합니다.
        # [2차]: 칸 번호를 알아냅니다.
        idx = self._get_bucket_index(key)
        # [1차]: 버킷 연결 리스트를 참조합니다.
        # [2차]: 서류철을 꺼냅니다.
        bucket = self._buckets[idx]

        # [1차]: 버킷 내 노드를 순회합니다.
        # [2차]: 봉투를 하나씩 넘겨봅니다.
        for node in bucket:
            # [1차]: 키가 일치하면 노드 객체 자체를 반환합니다.
            # [2차]: 찾은 서류 봉투를 그대로 건넵니다.
            if node.key == key:
                # [1차]: 노드를 반환합니다.
                # [2차]: 봉투를 전달합니다.
                return node
        # [1차]: 찾지 못하면 None을 반환합니다.
        # [2차]: 없으면 없다고 합니다.
        return None

    # [1차]: 주어진 키의 노드를 버킷 체인에서 O(1)으로 분리 제거합니다.
    # [2차]: 우편함 서류철에서 해당 이름표 봉투를 쏙 뽑아 영구 파기합니다.
    def remove(self, key):
        """
        키에 해당하는 항목을 삭제하고 기존 값을 반환합니다.
        존재하지 않으면 None을 반환합니다.
        """
        # [1차]: 버킷 인덱스를 계산합니다.
        # [2차]: 우편함 번호를 확인합니다.
        idx = self._get_bucket_index(key)
        # [1차]: 해당 버킷을 가져옵니다.
        # [2차]: 서류철을 꺼냅니다.
        bucket = self._buckets[idx]

        # [1차]: 버킷 내 노드를 순회합니다.
        # [2차]: 서류를 훑어봅니다.
        for node in bucket:
            # [1차]: 삭제 대상 키를 발견한 경우 처리합니다.
            # [2차]: 파기할 서류를 발견했습니다.
            if node.key == key:
                # [1차]: 삭제될 노드의 값을 임시 저장합니다.
                # [2차]: 버리기 전에 알맹이를 확인합니다.
                val = node.value
                # [1차]: DoublyLinkedList의 remove_node로 체인에서 제거합니다.
                # [2차]: 서류철 바인더에서 해당 봉투를 똑딱 떼어냅니다.
                bucket.remove_node(node)
                # [1차]: 해시맵 총 크기를 1 차감합니다.
                # [2차]: 총 서류 개수 장부를 1 줄입니다.
                self._size -= 1
                # [1차]: 제거된 기존 값을 반환합니다.
                # [2차]: 꺼낸 알맹이를 돌려줍니다.
                return val
        # [1차]: 키가 존재하지 않으면 None을 반환합니다.
        # [2차]: 버릴 게 없으니 그냥 끝냅니다.
        return None

    # [1차]: 해시맵 내에 특정 키가 존재하는지 여부를 불리언 값으로 반환합니다.
    # [2차]: "이 사람 서류가 우리 수납장에 있나요?"라는 물음에 네/아니오로 답합니다.
    def contains(self, key):
        """키 존재 여부를 반환합니다 (True/False)."""
        # [1차]: get 결과가 None이 아니거나 _key_exists가 참인지 판별합니다.
        # [2차]: 값이 None으로 저장된 경우까지 꼼꼼히 따져 존재 여부를 답합니다.
        return self.get(key) is not None or self._key_exists(key)

    # [1차]: 값이 None인 특수 엔트리까지 정확히 키 존재를 확인하는 보조 메서드입니다.
    # [2차]: 내용물이 비어있는 빈 봉투라도 서류철에 꽂혀 있는지 돋보기로 확인합니다.
    def _key_exists(self, key):
        """값이 None인 경우에도 키 존재 여부를 판별하기 위한 헬퍼"""
        # [1차]: 인덱스를 구합니다.
        # [2차]: 칸 번호를 봅니다.
        idx = self._get_bucket_index(key)
        # [1차]: 버킷을 가져옵니다.
        # [2차]: 서류철을 봅니다.
        bucket = self._buckets[idx]
        # [1차]: 노드를 순회하며 키 일치 여부를 검사합니다.
        # [2차]: 이름표가 있는지 넘겨봅니다.
        for node in bucket:
            # [1차]: 일치하면 True를 반환합니다.
            # [2차]: 있으면 네! 하고 외칩니다.
            if node.key == key:
                # [1차]: True 반환
                # [2차]: 존재 확인
                return True
        # [1차]: 없으면 False를 반환합니다.
        # [2차]: 없으면 아니오! 라고 답합니다.
        return False

    # [1차]: 해시맵에 보관된 모든 키 목록을 수집하여 파이썬 리스트로 반환합니다.
    # [2차]: 16개 우편함의 모든 서류철을 돌며 이름표만 싹 모아 명부를 만듭니다.
    def keys(self):
        """저장된 모든 키의 목록을 일반 파이썬 리스트로 반환합니다."""
        # [1차]: 키들을 모을 빈 리스트를 생성합니다.
        # [2차]: 백지 이름표 목록을 만듭니다.
        all_keys = []
        # [1차]: 모든 버킷 연결 리스트를 순회합니다.
        # [2차]: 0번 칸부터 마지막 칸까지 순서대로 둘러봅니다.
        for bucket in self._buckets:
            # [1차]: 각 버킷 내의 노드들을 순회합니다.
            # [2차]: 각 서류철에 꽂힌 봉투들을 봅니다.
            for node in bucket:
                # [1차]: 노드의 키를 결과 리스트에 추가합니다.
                # [2차]: 이름표 글자를 명부에 한 줄씩 적습니다.
                all_keys.append(node.key)
        # [1차]: 완성된 키 리스트를 반환합니다.
        # [2차]: 완성된 전체 명부를 전달합니다.
        return all_keys

    # [1차]: 해시맵에 저장된 총 고유 키의 개수를 반환합니다.
    # [2차]: 현재 수납장에 들어있는 총 서류 묶음 숫자를 알려줍니다.
    def size(self):
        """저장된 고유 키의 총 개수를 반환합니다."""
        # [1차]: _size 값을 반환합니다.
        # [2차]: 기록된 장부 숫자를 말합니다.
        return self._size

    # [1차]: 현재 할당된 총 버킷 배열의 크기를 반환합니다.
    # [2차]: 수납장에 설치된 우편함 칸수가 총 몇 칸인지 알려줍니다.
    def capacity(self):
        """현재 버킷 배열의 크기를 반환합니다."""
        # [1차]: _capacity 값을 반환합니다.
        # [2차]: 우편함 칸수를 반환합니다.
        return self._capacity

    # [1차]: 현재 로드 팩터(size / capacity)를 부동소수점 값으로 반환합니다.
    # [2차]: 수납장이 현재 몇 %나 차 있는지 포화도를 알려줍니다.
    def load_factor(self):
        """현재 해시맵의 로드 팩터를 반환합니다."""
        # [1차]: 크기를 용량으로 나눈 비율을 계산합니다.
        # [2차]: 서류 수 / 우편함 수 비율을 구합니다.
        return self._size / self._capacity

    # [1차]: 버킷 크기를 2배로 확장하고 기존 모든 노드를 재해싱하여 재배치합니다.
    # [2차]: 우편함 16칸이 꽉 찼으니 32칸짜리 새 수납장으로 서류를 전부 이사시킵니다.
    def _resize(self, new_capacity):
        """
        버킷 크기를 변경하고 모든 기존 항목을 재해싱(Rehashing)하여 재배치합니다.
        """
        # [1차]: 기존 버킷 배열 참조를 보관합니다.
        # [2차]: 헌 수납장의 서류들을 잠시 이삿짐 상자에 담아둡니다.
        old_buckets = self._buckets
        # [1차]: 용량을 새 크기로 업데이트합니다.
        # [2차]: 수납장 칸수 규격을 32칸으로 갱신합니다.
        self._capacity = new_capacity
        # [1차]: 새 크기의 빈 버킷 배열을 새로 생성합니다.
        # [2차]: 32칸짜리 깨끗한 새 우편함 수납장을 설치합니다.
        self._buckets = [DoublyLinkedList() for _ in range(self._capacity)]
        # [1차]: put 메서드 호출 시 크기가 다시 누적되므로 0으로 리셋합니다.
        # [2차]: 서류를 새로 꽂을 때마다 다시 셀 수 있게 장부를 0으로 비웁니다.
        self._size = 0  # put 호출 시 다시 1씩 증가

        # [1차]: 기존 버킷들을 순회합니다.
        # [2차]: 헌 수납장 칸들을 차례대로 비웁니다.
        for bucket in old_buckets:
            # [1차]: 기존 버킷 내의 노드들을 순회합니다.
            # [2차]: 서류 봉투들을 하나씩 꺼냅니다.
            for node in bucket:
                # [1차]: 새 버킷 크기에 맞춰 다시 해싱하여 삽입합니다.
                # [2차]: 새 32칸 우편함 규칙에 맞춰 서류를 알맞은 새 칸에 착착 꽂습니다.
                self.put(node.key, node.value)

# [1차]: 최소 힙(Min Heap) 모듈을 정의합니다.
# [2차]: 가장 작은 숫자(가장 임박한 만료 시각)가 언제나 꼭대기에 올라오는 피라미드 진열대를 만듭니다.
"""
최소 힙 (Min Heap) 모듈
dict, set, collections 없이 동적 배열(list)을 완전 이진 트리 구조로 사용하여 구현합니다.
TTL 만료 관리를 위한 (expire_at, key) 형태의 요소를 효율적으로 처리합니다.
"""

# [1차]: 완전 이진 트리 배열 기반의 MinHeap 클래스를 선언합니다.
# [2차]: 편의점 유통기한 우선 진열대 클래스입니다.
class MinHeap:
    """
    최소 힙 (Min-Heap)
    - 주요 메서드: push, pop, peek, size
    - 내부 복원 알고리즘: _heapify_up, _heapify_down
    """
    # [1차]: MinHeap 인스턴스를 초기화하며 빈 내부 리스트를 할당합니다.
    # [2차]: 피라미드 진열대 바닥판을 깨끗하게 비워둡니다.
    def __init__(self):
        # [1차]: 트리 노드들을 평면 배열로 저장할 _data 리스트를 초기화합니다.
        # [2차]: 피라미드 층별로 상품을 나열할 일렬 받침대를 둡니다.
        self._data = []

    # [1차]: 새 요소를 힙 맨 끝에 추가한 후 상향 힙 복원(_heapify_up)을 수행합니다.
    # [2차]: 새 물건을 진열대 맨 끝에 두고, 유통기한이 더 빠르면 위층으로 올려보냅니다.
    def push(self, item):
        """
        새 요소를 힙에 삽입하고 상향 힙 복원(_heapify_up)을 수행합니다.
        시간 복잡도: O(log N)
        """
        # [1차]: 내부 배열의 맨 마지막 위치에 item을 추가합니다.
        # [2차]: 진열대 맨 끝자리에 새 삼각김밥을 올려놓습니다.
        self._data.append(item)
        # [1차]: 새로 추가된 요소의 인덱스에서부터 _heapify_up을 시작합니다.
        # [2차]: 새 삼각김밥을 위층 김밥들과 비교하며 꼭대기를 향해 엘리베이터를 태웁니다.
        self._heapify_up(len(self._data) - 1)

    # [1차]: 힙의 최솟값(루트 노드)을 제거하고 반환하며 하향 힙 복원(_heapify_down)을 수행합니다.
    # [2차]: 진열대 꼭대기(가장 유통기한 임박)의 상품을 꺼내고 피라미드를 다시 정돈합니다.
    def pop(self):
        """
        최솟값(루트 노드)을 제거하고 반환합니다.
        힙이 비어있으면 None을 반환합니다.
        시간 복잡도: O(log N)
        """
        # [1차]: 힙이 비어있는지 확인합니다.
        # [2차]: 진열대에 상품이 하나도 없는지 봅니다.
        if self.is_empty():
            # [1차]: 비어있으면 None을 반환합니다.
            # [2차]: 줄 상품이 없으니 빈손으로 돌아옵니다.
            return None

        # [1차]: 요소가 1개뿐인 경우 pop으로 즉시 꺼내어 반환합니다.
        # [2차]: 딱 1개 남아있으면 뒤처리할 것도 없이 그냥 꺼내줍니다.
        if len(self._data) == 1:
            # [1차]: 유일한 요소를 꺼내 반환합니다.
            # [2차]: 하나뿐인 물건을 건넵니다.
            return self._data.pop()

        # [1차]: 0번 인덱스(루트)의 최솟값을 임시 보관합니다.
        # [2차]: 꼭대기 1등 상품을 따로 챙겨둡니다.
        root = self._data[0]
        # [1차]: 맨 끝의 요소를 꺼내어 루트 노드 위치에 덮어씁니다.
        # [2차]: 진열대 맨 밑바닥에 있던 상품을 꼭대기 빈자리로 휙 옮겨놓습니다.
        self._data[0] = self._data.pop()
        # [1차]: 루트 노드에서부터 하향 힙 복원(_heapify_down)을 수행합니다.
        # [2차]: 꼭대기로 올라온 상품을 아래층 상품들과 비교하며 알맞은 층까지 아래로 내립니다.
        self._heapify_down(0)
        # [1차]: 추출된 루트 요소를 반환합니다.
        # [2차]: 아까 챙겨둔 1등 상품을 손님에게 건넵니다.
        return root

    # [1차]: 루트 노드의 최솟값을 제거하지 않고 O(1) 시간에 확인합니다.
    # [2차]: 꼭대기에 있는 상품을 꺼내지 않고 유통기한만 슬쩍 곁눈질로 봅니다.
    def peek(self):
        """
        최솟값(루트 노드)을 제거하지 않고 확인합니다.
        힙이 비어있으면 None을 반환합니다.
        시간 복잡도: O(1)
        """
        # [1차]: 힙이 비어있으면 None을 반환합니다.
        # [2차]: 진열대가 비어있으면 아무것도 없다고 알립니다.
        if self.is_empty():
            # [1차]: None 반환
            # [2차]: 빈손 알림
            return None
        # [1차]: 0번 인덱스의 루트 요소를 반환합니다.
        # [2차]: 꼭대기 물건을 보여줍니다.
        return self._data[0]

    # [1차]: 힙에 저장된 총 요소의 개수를 반환합니다.
    # [2차]: 진열대 위에 놓인 총 상품 숫자를 알려줍니다.
    def size(self):
        """힙에 저장된 총 요소 개수를 반환합니다."""
        # [1차]: _data 리스트의 길이를 반환합니다.
        # [2차]: 물건 총 개수를 셉니다.
        return len(self._data)

    # [1차]: 힙이 비어있는지 여부를 불리언 값으로 반환합니다.
    # [2차]: 진열대에 물건이 0개인지 묻는 질문에 답합니다.
    def is_empty(self):
        """힙이 비어있는지 확인합니다."""
        # [1차]: 길이가 0인지 비교합니다.
        # [2차]: 텅 비었는지 확인합니다.
        return len(self._data) == 0

    # [1차]: 특정 인덱스의 요소를 부모와 비교하며 상향 이동시킵니다.
    # [2차]: 상품이 위층 상품보다 더 긴급하면 윗자리와 맞바꾸며 엘리베이터를 타고 올라갑니다.
    def _heapify_up(self, index):
        """
        새로 삽입된 요소를 부모와 비교하며 적절한 위치까지 위로 끌어올립니다.
        부모 인덱스: (index - 1) // 2
        """
        # [1차]: 루트(0번)에 도달할 때까지 상향 순회합니다.
        # [2차]: 꼭대기 층에 닿을 때까지 계속 올라가 봅니다.
        while index > 0:
            # [1차]: 부모 노드의 인덱스를 계산합니다: (index - 1) // 2
            # [2차]: 내 바로 윗층 부모 칸 번호를 계산합니다.
            parent_idx = (index - 1) // 2
            # [1차]: 현재 노드가 부모 노드보다 값이 작은지(우선순위가 높은지) 비교합니다.
            # [2차]: 내 유통기한이 윗층 물건보다 더 임박했는지 비교합니다.
            if self._data[index] < self._data[parent_idx]:
                # [1차]: 자식 노드와 부모 노드의 위치를 스왑(맞교환)합니다.
                # [2차]: 더 급한 내 물건을 윗층으로 올리고 윗층 물건을 아래로 내립니다.
                self._data[index], self._data[parent_idx] = self._data[parent_idx], self._data[index]
                # [1차]: 인덱스를 부모 위치로 갱신하여 상향 추적을 이어갑니다.
                # [2차]: 내 위치가 윗층으로 올라갔으니 그 위치에서 다시 검사를 준비합니다.
                index = parent_idx
            else:
                # [1차]: 부모보다 작지 않으면 힙 불변성이 만족되므로 루프를 탈출합니다.
                # [2차]: 윗층 물건이 나보다 더 급한 물건이니 현재 층에 얌전히 멈춥니다.
                break

    # [1차]: 특정 인덱스의 요소를 자식들과 비교하며 하향 이동시킵니다.
    # [2차]: 꼭대기로 올라온 엉뚱한 상품을 아래층 더 급한 상품들과 비교하며 알맞은 층으로 내립니다.
    def _heapify_down(self, index):
        """
        루트 또는 특정 위치의 요소를 자식들과 비교하며 적절한 위치까지 아래로 내립니다.
        좌측 자식: 2 * index + 1
        우측 자식: 2 * index + 2
        """
        # [1차]: 배열의 총 길이를 캐싱합니다.
        # [2차]: 진열대 바닥 칸 번호 한계를 적어둡니다.
        length = len(self._data)
        # [1차]: 적절한 위치를 찾을 때까지 무한 루프로 하향 탐색합니다.
        # [2차]: 제자리를 찾을 때까지 에스컬레이터를 타고 내려갑니다.
        while True:
            # [1차]: 좌측 자식의 인덱스를 계산합니다: 2 * index + 1
            # [2차]: 왼쪽 아래 자식 칸 번호를 계산합니다.
            left_child_idx = 2 * index + 1
            # [1차]: 우측 자식의 인덱스를 계산합니다: 2 * index + 2
            # [2차]: 오른쪽 아래 자식 칸 번호를 계산합니다.
            right_child_idx = 2 * index + 2
            # [1차]: 현재 자신과 두 자식 중 가장 작은 요소의 인덱스를 추적합니다.
            # [2차]: 나, 왼쪽 자식, 오른쪽 자식 셋 중 유통기한 1등을 찾기 위한 표식입니다.
            smallest_idx = index

            # [1차]: 좌측 자식이 존재하고 현재 최솟값보다 더 작으면 갱신합니다.
            # [2차]: 왼쪽 자식이 존재하고 나보다 유통기한이 더 급하면 1등 후보로 지목합니다.
            if left_child_idx < length and self._data[left_child_idx] < self._data[smallest_idx]:
                # [1차]: smallest_idx를 좌측 자식 인덱스로 변경합니다.
                # [2차]: 왼쪽 자식을 1등 후보로 바꿉니다.
                smallest_idx = left_child_idx

            # [1차]: 우측 자식이 존재하고 현재 최솟값보다 더 작으면 갱신합니다.
            # [2차]: 오른쪽 자식이 존재하고 지금까지의 1등 후보보다 더 급하면 새 1등으로 지목합니다.
            if right_child_idx < length and self._data[right_child_idx] < self._data[smallest_idx]:
                # [1차]: smallest_idx를 우측 자식 인덱스로 변경합니다.
                # [2차]: 오른쪽 자식을 1등 후보로 바꿉니다.
                smallest_idx = right_child_idx

            # [1차]: 가장 작은 요소가 자기 자신이 아니라면 스왑하고 계속 내려갑니다.
            # [2차]: 자식들 중에 나보다 더 급한 물건이 있다면 자리를 맞바꿔주고 더 내려갑니다.
            if smallest_idx != index:
                # [1차]: 현재 노드와 smallest_idx 노드를 스왑합니다.
                # [2차]: 더 급한 자식 상품을 위로 올리고 내 상품을 아래로 내립니다.
                self._data[index], self._data[smallest_idx] = self._data[smallest_idx], self._data[index]
                # [1차]: 인덱스를 smallest_idx로 갱신하여 하향 순회를 이어갑니다.
                # [2차]: 내려간 새 층에서 다음 자식들과 다시 비교를 준비합니다.
                index = smallest_idx
            else:
                # [1차]: 자신이 자식들보다 모두 작거나 같으면 힙 정렬이 완성되었으므로 종료합니다.
                # [2차]: 내가 자식들보다 더 급한 물건이거나 바닥에 도착했으니 이 층에 자리를 잡습니다.
                break

# [1차]: LRU 캐시 매니저 모듈을 정의합니다.
# [2차]: 최근에 본 책은 책상 맨 위에 올려두고 가장 오래 안 본 책은 밑바닥으로 보내는 독서대를 만듭니다.
"""
LRU 추적 관리자 (LRU Cache Manager) 모듈
DoublyLinkedList와 자체 HashMap을 결합하여 O(1) 시간 복잡도로 키 접근 추적 및 최하위(LRU) 키 방출을 수행합니다.
dict, set, collections는 일절 사용하지 않습니다.
"""

# [1차]: O(1) 순서 재배치를 위한 DoublyLinkedList 및 Node 클래스를 가져옵니다.
# [2차]: 책들을 엮어둘 양방향 기차 대형 부품들을 챙겨옵니다.
from src.doubly_linked_list import DoublyLinkedList, Node
# [1차]: O(1) 키 탐색을 위한 자체 HashMap 클래스를 가져옵니다.
# [2차]: 책 이름표를 보고 즉시 책 위치를 찍어줄 번호표 수납장을 가져옵니다.
from src.hash_map import HashMap


# [1차]: LRUManager 클래스를 정의합니다.
# [2차]: 가장 오래 안 쓴 물건을 칼같이 골라내는 스마트 독서대 관리인입니다.
class LRUManager:
    """
    O(1) LRU(Least Recently Used) 추적 관리자
    - 최근 접근된 키는 리스트의 Head 쪽에 위치
    - 가장 오랫동안 접근되지 않은 키는 리스트의 Tail 쪽에 위치
    - 자체 HashMap(key -> Node)을 통해 O(1) 노드 탐색 및 위치 갱신
    """
    # [1차]: LRUManager 인스턴스를 초기화하고 리스트와 노드 맵을 생성합니다.
    # [2차]: 순서를 기억할 트랙(리스트)과 책 위치를 적어둘 장부(해시맵)를 준비합니다.
    def __init__(self):
        # [1차]: 접근 순서를 기록하는 DoublyLinkedList 인스턴스를 할당합니다.
        # [2차]: 최근 순서대로 책을 꽂아둘 양방향 책꽂이를 놓습니다.
        self._list = DoublyLinkedList()
        # [1차]: key에서 Node 객체 참조를 매핑하는 HashMap 인스턴스를 할당합니다.
        # [2차]: "이 책이 책꽂이 몇 번째 칸에 꽂혀 있는지" 적어둔 색인 장부를 엽니다.
        self._node_map = HashMap()

    # [1차]: 특정 키에 대한 접근(Read/Write)을 기록하고 해당 노드를 맨 앞으로 이동시킵니다.
    # [2차]: 손님이 책을 읽거나 새로 사 오면, 그 책을 책꽂이 맨 앞자리(1등석)로 옮깁니다.
    def access(self, key):
        """
        키에 접근(조회 또는 갱신/추가)했을 때 해당 키를 가장 최근(Head)으로 갱신합니다.
        O(1) 연산
        """
        # [1차]: HashMap에서 해당 키에 대응하는 기존 Node 참조를 O(1)으로 조회합니다.
        # [2차]: 색인 장부를 보고 이 책이 이미 책꽂이에 꽂혀 있는지 확인합니다.
        existing_node = self._node_map.get(key)
        # [1차]: 기존 노드가 존재하는 경우 분기합니다.
        # [2차]: 이미 책꽂이에 꽂혀 있던 책인 경우입니다.
        if existing_node is not None:
            # [1차]: DoublyLinkedList의 move_to_front를 호출하여 O(1)으로 헤드 바로 뒤로 이동합니다.
            # [2차]: 책꽂이 중간에서 그 책을 쏙 빼서 맨 앞자리로 탁 꽂아 넣습니다.
            self._list.move_to_front(existing_node)
        else:
            # [1차]: 신규 키인 경우 새 Node를 생성합니다.
            # [2차]: 오늘 서점에서 처음 사 온 새 책인 경우입니다.
            new_node = Node(key=key, value=None, data=key)
            # [1차]: 리스트의 맨 앞에 새 노드를 O(1)으로 삽입합니다.
            # [2차]: 책꽂이 맨 앞 1등 자리에 새 책을 꽂습니다.
            self._list.insert_front(new_node)
            # [1차]: HashMap에 key -> new_node 매핑을 등록합니다.
            # [2차]: 색인 장부에 "이 새 책은 여기에 있어요" 하고 위치를 적어둡니다.
            self._node_map.put(key, new_node)

    # [1차]: 키가 삭제되거나 만료되었을 때 LRU 추적 구조에서 O(1)으로 제거합니다.
    # [2차]: 책을 버리거나 분실했을 때 책꽂이와 색인 장부에서 깨끗이 지웁니다.
    def remove(self, key):
        """
        키가 삭제되었거나 만료되었을 때 LRU 추적 구조에서 O(1)으로 제거합니다.
        """
        # [1차]: HashMap에서 해당 키의 Node를 조회합니다.
        # [2차]: 색인 장부에서 책 위치를 찾습니다.
        node = self._node_map.get(key)
        # [1차]: 노드가 존재하면 리스트와 맵에서 동시에 제거합니다.
        # [2차]: 책이 실제로 꽂혀 있다면 제거 절차를 밟습니다.
        if node is not None:
            # [1차]: 연결 리스트에서 해당 노드를 O(1)으로 분리합니다.
            # [2차]: 책꽂이에서 그 책을 쏙 빼냅니다.
            self._list.remove_node(node)
            # [1차]: HashMap에서 해당 키의 매핑을 제거합니다.
            # [2차]: 색인 장부에서 해당 책 이름을 지웁니다.
            self._node_map.remove(key)
            # [1차]: 성공을 의미하는 True를 반환합니다.
            # [2차]: 잘 치웠다고 보고합니다.
            return True
        # [1차]: 키가 없으면 False를 반환합니다.
        # [2차]: 원래 없던 책이니 지울 게 없다고 보고합니다.
        return False

    # [1차]: 가장 오랫동안 접근되지 않은(리스트 tail 바로 앞) 노드를 O(1)으로 꺼내어 제거하고 key를 반환합니다.
    # [2차]: 책꽂이 맨 밑바닥에 먼지 쌓인 가장 오래된 책을 1초 만에 꺼내어 버립니다.
    def pop_lru(self):
        """
        가장 오랫동안 사용되지 않은(Tail에 위치한) 키를 O(1)으로 추출하고 제거합니다.
        추출된 key를 반환하며, 비어있으면 None을 반환합니다.
        """
        # [1차]: 리스트가 비어있는지 확인합니다.
        # [2차]: 책꽂이에 책이 한 권도 없는지 봅니다.
        if self._list.is_empty():
            # [1차]: 비어있으면 None을 반환합니다.
            # [2차]: 버릴 책이 없으니 빈손으로 돌아옵니다.
            return None

        # [1차]: 리스트의 맨 뒤 유효 노드를 remove_back으로 O(1) 제거합니다.
        # [2차]: 책꽂이 맨 꼴찌 자리에 꽂혀 있던 가장 오래된 책을 뺍니다.
        lru_node = self._list.remove_back()
        # [1차]: 제거된 노드가 존재하면 HashMap에서도 키를 정리하고 key를 반환합니다.
        # [2차]: 꺼낸 책이 있다면 장부에서도 지우고 그 책 이름을 알려줍니다.
        if lru_node is not None:
            # [1차]: HashMap에서 해당 노드의 key 매핑을 제거합니다.
            # [2차]: 색인 장부에서 그 책의 이름을 지웁니다.
            self._node_map.remove(lru_node.key)
            # [1차]: 방출된 노드의 key를 반환합니다.
            # [2차]: "이 책을 방출했습니다" 하고 책 이름을 알려줍니다.
            return lru_node.key
        # [1차]: 비정상 상태 시 None을 반환합니다.
        # [2차]: 아무것도 못 꺼냈으면 빈손으로 끝냅니다.
        return None

    # [1차]: 현재 LRU 추적 중인 총 키 개수를 반환합니다.
    # [2차]: 현재 책꽂이에 꽂혀 있는 책이 총 몇 권인지 세어줍니다.
    def size(self):
        """현재 추적 중인 키 개수 반환"""
        # [1차]: 내부 리스트의 size를 반환합니다.
        # [2차]: 책꽂이의 책 권수를 알려줍니다.
        return self._list.size()

# [1차]: 메모리 관리자(Memory Manager) 모듈을 정의합니다.
# [2차]: 책가방의 허용 무게(바이트 용량)를 재고, 가방이 터지려고 하면 책을 버려주는 스마트 저울을 만듭니다.
"""
메모리 관리자 (Memory Manager) 모듈
used_memory(UTF-8 바이트 총합)와 maxmemory 한도를 추적하고,
메모리 한도 초과 시 LRU 정책에 따라 키를 자동 방출(Eviction)합니다.
"""

# [1차]: MemoryManager 클래스를 선언합니다.
# [2차]: 매 순간 메모리 사용량을 바이트 단위로 달아보는 저울 관리인입니다.
class MemoryManager:
    """
    메모리 관리 및 한도 검사 클래스
    - used_memory: Σ(len(utf8(key)) + len(utf8(value)))
    - maxmemory: 0은 무제한, 양수는 바이트 단위 제한
    - evicted_keys: 메모리 초과로 인해 제거된 키의 누적 개수
    """
    # [1차]: MemoryManager 인스턴스를 초기화하며 maxmemory를 설정합니다.
    # [2차]: 가방의 최대 허용 무게를 입력받고 저울 바늘을 0으로 맞춥니다.
    def __init__(self, maxmemory=0):
        # [1차]: 최대 메모리 제한(바이트)을 인스턴스 변수에 저장합니다 (0: 무제한).
        # [2차]: 최대 무게 한도를 적어둡니다 (0이면 얼마든지 담아도 됨).
        self._maxmemory = maxmemory
        # [1차]: 현재 사용 중인 총 UTF-8 바이트 수를 0으로 초기화합니다.
        # [2차]: 현재 가방에 담긴 총 무게를 0g으로 시작합니다.
        self._used_memory = 0
        # [1차]: 메모리 초과로 강제 방출(evict)된 누적 키 개수를 0으로 초기화합니다.
        # [2차]: 지금까지 가방이 무거워서 밖으로 던져버린 책의 총 권수를 0권으로 잡습니다.
        self._evicted_keys = 0

    # [1차]: 키와 값의 문자열을 UTF-8 바이트로 인코딩한 길이의 합을 계산하는 정적 메서드입니다.
    # [2차]: 물건과 이름표를 전자저울에 올려 총 몇 g(바이트)인지 무게를 측정합니다.
    @staticmethod
    def calc_entry_size(key, value):
        """키와 값의 UTF-8 바이트 크기 합산 계산"""
        # [1차]: key를 문자열로 변환한 후 UTF-8 바이트 배열로 인코딩합니다.
        # [2차]: 이름표를 포장해서 무게를 잽니다.
        key_bytes = str(key).encode("utf-8")
        # [1차]: value를 문자열로 변환한 후 UTF-8 바이트 배열로 인코딩합니다.
        # [2차]: 내용물을 포장해서 무게를 잽니다.
        val_bytes = str(value).encode("utf-8")
        # [1차]: 키 바이트 수와 값 바이트 수의 합을 반환합니다.
        # [2차]: 둘의 무게를 합친 총 무게를 알려줍니다.
        return len(key_bytes) + len(val_bytes)

    # [1차]: maxmemory 속성의 getter 메서드입니다.
    # [2차]: 현재 가방의 무게 제한이 몇 g인지 물어봅니다.
    @property
    def maxmemory(self):
        # [1차]: _maxmemory 값을 반환합니다.
        # [2차]: 한도 무게를 말해줍니다.
        return self._maxmemory

    # [1차]: maxmemory 속성의 setter 메서드로 0 이상의 정수만 안전하게 할당합니다.
    # [2차]: 가방의 무게 한도를 새 값으로 조절합니다 (음수는 0으로 보정).
    @maxmemory.setter
    def maxmemory(self, value):
        # [1차]: 정수형으로 변환 후 0 이상의 값으로 클램핑합니다.
        # [2차]: 음수가 들어오지 못하게 0으로 딱 막아줍니다.
        self._maxmemory = max(0, int(value))

    # [1차]: used_memory 속성의 getter 메서드입니다.
    # [2차]: 현재 가방이 총 몇 g인지 저울 눈금을 읽습니다.
    @property
    def used_memory(self):
        # [1차]: _used_memory 값을 반환합니다.
        # [2차]: 현재 사용 중인 바이트 무게를 알려줍니다.
        return self._used_memory

    # [1차]: evicted_keys 속성의 getter 메서드입니다.
    # [2차]: 쫓겨난 책이 지금까지 총 몇 권인지 물어봅니다.
    @property
    def evicted_keys(self):
        # [1차]: _evicted_keys 값을 반환합니다.
        # [2차]: 쫓겨난 책 권수를 말해줍니다.
        return self._evicted_keys

    # [1차]: 단일 엔트리 자체의 크기가 maxmemory를 초과하는지 OOM 여부를 판별합니다.
    # [2차]: 책 한 권의 무게가 가방 전체 한도보다 더 무거워서 아예 안 들어가는지 검사합니다.
    def is_single_entry_oom(self, key, value):
        """
        단일 엔트리 자체의 크기가 maxmemory를 초과하는지 검사합니다.
        maxmemory > 0이고 단일 크기 > maxmemory이면 True (저장 불가 OOM)
        """
        # [1차]: maxmemory가 0 이하이면 무제한이므로 OOM이 발생하지 않습니다.
        # [2차]: 가방 크기가 무제한이면 어떤 거대한 책이 와도 다 들어갑니다.
        if self._maxmemory <= 0:
            # [1차]: False 반환
            # [2차]: 문제없음 보고
            return False
        # [1차]: 키와 값의 크기를 바이트 단위로 계산합니다.
        # [2차]: 이번 책 한 권의 무게를 잽니다.
        entry_size = self.calc_entry_size(key, value)
        # [1차]: 엔트리 크기가 maxmemory보다 크면 True를 반환합니다.
        # [2차]: 책 한 권 무게가 가방 전체 한도를 넘으면 가방 터짐(True)을 알립니다.
        return entry_size > self._maxmemory

    # [1차]: 새 엔트리가 추가될 때 used_memory를 증가시킵니다.
    # [2차]: 새 책을 가방에 넣었으니 그만큼 저울 눈금을 올립니다.
    def add_entry_memory(self, key, value):
        """새 엔트리 추가 시 메모리 사용량 증가"""
        # [1차]: 새 엔트리의 UTF-8 바이트 크기를 used_memory에 가산합니다.
        # [2차]: 저울 무게에 새 책 무게를 더합니다.
        self._used_memory += self.calc_entry_size(key, value)

    # [1차]: 기존 엔트리가 덮어쓰여질 때 값의 바이트 차이만큼 used_memory를 보정합니다.
    # [2차]: 책 속의 얇은 페이지를 두꺼운 새 페이지로 교체했을 때 무게 차이만큼 저울을 조정합니다.
    def update_entry_memory(self, key, old_value, new_value):
        """기존 엔트리 갱신 시 값 차이만큼 메모리 사용량 반영"""
        # [1차]: 기존 값의 UTF-8 바이트 길이를 계산합니다.
        # [2차]: 옛날 페이지 무게를 잽니다.
        old_val_size = len(str(old_value).encode("utf-8"))
        # [1차]: 새 값의 UTF-8 바이트 길이를 계산합니다.
        # [2차]: 새 페이지 무게를 잽니다.
        new_val_size = len(str(new_value).encode("utf-8"))
        # [1차]: (새 크기 - 기존 크기) 차이만큼 used_memory에 반영합니다.
        # [2차]: 무게 차이만큼 저울 눈금을 올려주거나 내려줍니다.
        self._used_memory += (new_val_size - old_val_size)

    # [1차]: 엔트리가 삭제될 때 used_memory를 차감합니다.
    # [2차]: 책을 가방에서 뺐으니 그만큼 저울 눈금을 내립니다.
    def remove_entry_memory(self, key, value):
        """엔트리 삭제 시 메모리 사용량 차감"""
        # [1차]: 삭제 대상의 바이트 크기를 계산합니다.
        # [2차]: 꺼낸 책의 무게를 잽니다.
        size = self.calc_entry_size(key, value)
        # [1차]: used_memory에서 차감하며 음수가 되지 않도록 max(0, ...) 처리합니다.
        # [2차]: 무게를 빼되, 저울이 마이너스가 되지 않도록 0에서 딱 멈춥니다.
        self._used_memory = max(0, self._used_memory - size)

    # [1차]: maxmemory 한도 초과 시 한도 이하가 될 때까지 LRU 키를 지속 방출합니다.
    # [2차]: 가방이 터질 때까지 차오르면, 가방 무게가 정상 범위가 될 때까지 가장 오래 안 본 책부터 밖으로 던집니다.
    def evict_if_needed(self, lru_manager, del_callback):
        """
        maxmemory > 0이고 used_memory > maxmemory인 경우,
        used_memory <= maxmemory가 될 때까지 LRU 키를 방출합니다.
        del_callback(key)은 주 저장소 및 TTL 등에서 해당 키를 완전 정리하는 콜백입니다.
        """
        # [1차]: maxmemory가 0 이하이면 무제한이므로 방출 루프를 수행하지 않습니다.
        # [2차]: 가방 용량이 무제한이면 책을 버릴 필요가 없습니다.
        if self._maxmemory <= 0:
            # [1차]: 함수를 즉시 종료합니다.
            # [2차]: 그대로 둡니다.
            return

        # [1차]: used_memory가 maxmemory를 초과하는 동안 반복 실행합니다.
        # [2차]: 가방 무게가 한도 무게보다 무거운 동안 계속 반복합니다.
        while self._used_memory > self._maxmemory:
            # [1차]: LRUManager에서 가장 오래된 키를 O(1)으로 pop합니다.
            # [2차]: 책꽂이 맨 밑바닥에서 가장 오래 안 읽은 책을 1권 꺼냅니다.
            lru_key = lru_manager.pop_lru()
            # [1차]: 더 이상 방출할 키가 없으면 루프를 탈출합니다.
            # [2차]: 더 이상 꺼낼 책이 없으면 멈춥니다.
            if lru_key is None:
                # [1차]: break로 반복문 탈출
                # [2차]: 루프 종료
                break
            
            # [1차]: 콜백을 호출하여 저장소, TTL, 메모리에서 해당 키를 일괄 정리합니다.
            # [2차]: 집주인에게 "이 책 버려주세요" 하고 방출 처리를 맡깁니다.
            del_callback(lru_key, is_eviction=True)
            # [1차]: 방출된 키 카운터를 1 증가시킵니다.
            # [2차]: 쫓겨난 책 권수 카운터를 1 올립니다.
            self._evicted_keys += 1

    # [1차]: INFO memory 명령어 요구 규격에 맞추어 포맷팅된 문자열을 반환합니다.
    # [2차]: 현재 가방 상태(사용 무게, 한도 무게, 버린 책 수)를 깔끔한 영수증으로 뽑아줍니다.
    def get_info_memory(self):
        """INFO memory 명령어용 출력 문자열 생성"""
        # [1차]: used_memory, maxmemory, evicted_keys 세 항목을 개행 문자로 연결하여 반환합니다.
        # [2차]: 3대 핵심 메모리 지표를 알기 쉽게 세 줄로 적어줍니다.
        return f"used_memory:{self._used_memory}\nmaxmemory:{self._maxmemory}\nevicted_keys:{self._evicted_keys}"

# [1차]: 최소 힙 기반 TTL(Time To Live) 관리자 모듈을 정의합니다.
# [2차]: 유통기한이 적힌 삼각김밥들을 최소 힙 진열대에 올리고 지연 폐기를 관리하는 유통기한 전담반을 만듭니다.
"""
TTL(Time To Live) 관리자 모듈
MinHeap과 자체 HashMap을 결합하여 (expire_at, key) 만료 시간을 관리하고 지연 삭제(Lazy Deletion)를 수행합니다.
dict, set, collections는 사용하지 않습니다.
"""

# [1차]: 현재 시각 측정을 위한 표준 time 모듈을 임포트합니다.
# [2차]: 매장 벽에 걸린 정확한 전자시계를 쳐다봅니다.
import time
# [1차]: 남은 시간 올림(ceil) 계산을 위한 표준 math 모듈을 임포트합니다.
# [2차]: 남은 초가 0.1초라도 있으면 1초로 계산해 주는 친절한 올림 계산기를 챙깁니다.
import math
# [1차]: 만료 시각 우선순위 관리를 위한 MinHeap 클래스를 임포트합니다.
# [2차]: 유통기한 빠른 순으로 정렬해 주는 피라미드 진열대를 가져옵니다.
from src.min_heap import MinHeap
# [1차]: 키별 최신 만료 시각 매핑을 위한 자체 HashMap 클래스를 임포트합니다.
# [2차]: 상품별 유통기한 메모 장부(해시맵)를 가져옵니다.
from src.hash_map import HashMap


# [1차]: TTLManager 클래스를 선언합니다.
# [2차]: 유통기한 확인 및 만료 폐기를 전문으로 지휘하는 매니저입니다.
class TTLManager:
    """
    최소 힙 기반 TTL 관리자
    - _heap: MinHeap((expire_at, key)) -> O(log N) 만료 시간 우선순위 관리
    - _expire_map: HashMap(key -> expire_at) -> O(1) 만료 시간 조회 및 유효성 검증
    """
    # [1차]: TTLManager 인스턴스를 초기화하며 힙, 해시맵, 시간 제공자를 설정합니다.
    # [2차]: 진열대, 메모 장부, 전자시계를 준비합니다.
    def __init__(self, time_provider=time.time):
        # [1차]: (expire_at, key) 튜플을 저장할 MinHeap 인스턴스를 생성합니다.
        # [2차]: 유통기한 삼각김밥 피라미드 진열대를 만듭니다.
        self._heap = MinHeap()
        # [1차]: key -> expire_at 매핑을 저장할 HashMap 인스턴스를 생성합니다.
        # [2차]: 각 상품의 최신 유통기한을 기록해둘 장부를 만듭니다.
        self._expire_map = HashMap()
        # [1차]: 현재 시각을 반환하는 함수 포인터를 저장합니다 (테스트 시 가상 시간 주입 가능).
        # [2차]: 진짜 시계 또는 테스트용 연습 시계를 지정합니다.
        self._time_provider = time_provider

    # [1차]: 특정 키의 만료 시간(초)을 설정하고 힙과 맵에 기록합니다.
    # [2차]: 상품에 "지금부터 N초 뒤 폐기"라는 딱지를 붙이고 진열대에 올립니다.
    def set_ttl(self, key, seconds):
        """
        키의 만료 시간(초)을 설정합니다.
        seconds <= 0이면 즉시 만료 상태로 간주합니다.
        """
        # [1차]: 현재 시각을 구합니다.
        # [2차]: 지금 몇 시 몇 분 몇 초인지 시계를 봅니다.
        now = self._time_provider()
        # [1차]: 만료 시각 타임스탬프(now + seconds)를 계산합니다.
        # [2차]: 유통기한 만료 시각을 계산합니다.
        expire_at = now + seconds
        # [1차]: _expire_map에 key -> expire_at 매핑을 갱신합니다.
        # [2차]: 장부에 이 상품의 최신 만료 시각을 적습니다.
        self._expire_map.put(key, expire_at)
        # [1차]: 힙에 (expire_at, key) 튜플을 push합니다.
        # [2차]: 진열대 위에 이 김밥을 올리고 알맞은 층으로 정돈시킵니다.
        self._heap.push((expire_at, key))
        # [1차]: 계산된 만료 타임스탬프를 반환합니다.
        # [2차]: 정해진 만료 시각을 알려줍니다.
        return expire_at

    # [1차]: 키의 만료 시간 설정을 해제하고 맵에서 제거합니다.
    # [2차]: 상품의 유통기한 딱지를 떼어내어 평생 보관 가능한 무기한 상품으로 바꿉니다.
    def remove_ttl(self, key):
        """키의 TTL 설정을 제거합니다 (SET 덮어쓰기나 DEL 호출 시 사용)."""
        # [1차]: _expire_map에서 해당 key를 삭제합니다.
        # [2차]: 장부에서 유통기한 기록을 깨끗이 지웁니다.
        return self._expire_map.remove(key)

    # [1차]: 키의 만료 타임스탬프를 조회합니다.
    # [2차]: 이 상품의 유통기한 시각이 언제로 적혀 있는지 장부를 봅니다.
    def get_expire_at(self, key):
        """키의 만료 타임스탬프를 조회합니다 (미설정 시 None)."""
        # [1차]: _expire_map에서 만료 시각을 조회하여 반환합니다.
        # [2차]: 장부에 적힌 만료 시각을 읽어줍니다.
        return self._expire_map.get(key)

    # [1차]: 키가 만료 시각을 초과했는지 여부를 불리언 값으로 확인합니다.
    # [2차]: 지금 시각이 유통기한을 지났는지 따져보고 상했는지 판별합니다.
    def is_expired(self, key):
        """
        키가 만료되었는지 여부를 확인합니다.
        만료시간이 설정되어 있고 현재 시각 >= expire_at 이면 True를 반환합니다.
        """
        # [1차]: 해당 키의 만료 시각을 가져옵니다.
        # [2차]: 장부에서 유통기한을 찾습니다.
        expire_at = self._expire_map.get(key)
        # [1차]: 만료 시각이 설정되지 않았다면 만료되지 않은 것입니다.
        # [2차]: 유통기한 딱지가 없는 물건은 영구 보관이므로 안 상했습니다.
        if expire_at is None:
            # [1차]: False 반환
            # [2차]: 안전함 보고
            return False
        # [1차]: 현재 시각이 expire_at 이상인지 비교합니다.
        # [2차]: 지금 시계 바늘이 유통기한 시각을 넘어섰는지 봅니다.
        return self._time_provider() >= expire_at

    # [1차]: Redis TTL 명령어 규격에 맞추어 남은 초(-2, -1, N)를 계산하여 반환합니다.
    # [2차]: 손님이 "이 김밥 며칠 남았어요?" 물어봤을 때 정확한 숫자로 답해줍니다.
    def get_ttl_seconds(self, key, key_exists_in_storage):
        """
        Redis 규격에 따른 TTL 반환값:
        - key가 없으면: -2
        - key는 존재하지만 만료 시간이 없으면: -1
        - 만료 시간이 있고 아직 유효하면: 남은 초 (정수 N, math.ceil 또는 round)
        """
        # [1차]: 저장소 자체에 키가 없으면 -2를 반환합니다.
        # [2차]: 가게에 아예 없는 물건이면 -2라고 답합니다.
        if not key_exists_in_storage:
            # [1차]: -2 반환
            # [2차]: 물건 없음
            return -2

        # [1차]: 만료 시각을 맵에서 조회합니다.
        # [2차]: 장부에서 유통기한을 찾습니다.
        expire_at = self._expire_map.get(key)
        # [1차]: 만료 시각이 설정되어 있지 않은 영구 키인 경우 -1을 반환합니다.
        # [2차]: 유통기한이 없는 영구 보관 상품이면 -1이라고 답합니다.
        if expire_at is None:
            # [1차]: -1 반환
            # [2차]: 무기한 상품
            return -1

        # [1차]: 현재 시각을 구합니다.
        # [2차]: 지금 시각을 봅니다.
        now = self._time_provider()
        # [1차]: 잔여 초(expire_at - now)를 계산합니다.
        # [2차]: 유통기한까지 남은 시간을 계산합니다.
        remaining = expire_at - now
        # [1차]: 이미 시간이 지난 경우 만료(-2)로 처리합니다.
        # [2차]: 0초 이하로 이미 지난 물건은 만료(-2) 처리합니다.
        if remaining <= 0:
            # [1차]: -2 반환
            # [2차]: 이미 상함
            return -2  # 이미 만료됨
        # [1차]: 1초 미만이라도 남아있으면 최소 1 이상으로 올림하여 반환합니다.
        # [2차]: 0.5초 남았어도 올림해서 1초 남았다고 친절하게 알려줍니다.
        return max(1, math.ceil(remaining))

    # [1차]: 최소 힙의 루트를 확인하여 만료된 항목들을 능동적으로 pop하고 콜백으로 삭제합니다.
    # [2차]: 진열대 꼭대기를 보며 유통기한 지난 상품들을 차례대로 쓸어 담아 쓰레기통에 버립니다.
    def cleanup_expired(self, del_callback):
        """
        힙의 루트를 확인하여 현재 시각 기준 이미 만료된 키들을 능동적으로 정리합니다.
        유효하지 않거나 이미 갱신된 stale entry는 자동으로 버립니다.
        """
        # [1차]: 현재 시각을 측정합니다.
        # [2차]: 지금 시각을 확인합니다.
        now = self._time_provider()
        # [1차]: 힙이 빌 때까지 루트를 검사하는 루프를 실행합니다.
        # [2차]: 진열대가 빌 때까지 꼭대기 상품들을 살펴봅니다.
        while not self._heap.is_empty():
            # [1차]: 루트 요소를 제거하지 않고 peek으로 봅니다.
            # [2차]: 진열대 꼭대기 1등 상품의 유통기한을 슬쩍 봅니다.
            peek_item = self._heap.peek()
            # [1차]: None인 경우 중단합니다.
            # [2차]: 없으면 그만둡니다.
            if peek_item is None:
                # [1차]: 루프 탈출
                # [2차]: 중단
                break
            # [1차]: 튜플에서 expire_at과 key를 분해합니다.
            # [2차]: 만료 시각과 상품 이름을 분리해 봅니다.
            expire_at, key = peek_item
            # [1차]: 가장 빠른 만료 시각조차 아직 현재 시각보다 미래라면 루프를 종료합니다.
            # [2차]: 꼭대기 김밥도 아직 유통기한이 안 지났다면, 아래층 김밥들은 볼 것도 없이 전부 안전하니 청소를 끝냅니다!
            if expire_at > now:
                # [1차]: 추가 탐색 없이 즉시 종료
                # [2차]: 1초 만에 확인 완료
                break

            # [1차]: 만료 시각이 도래한 요소를 힙에서 pop합니다.
            # [2차]: 지난 상품을 진열대에서 뺍니다.
            self._heap.pop()

            # [1차]: 해당 키의 최신 expire_at이 현재 튜플의 expire_at과 일치하는지 유효성을 검증합니다.
            # [2차]: 장부를 펼쳐 이 상품이 그사이 새 유통기한으로 갱신된 건 아닌지 확인합니다.
            current_expire_at = self._expire_map.get(key)
            # [1차]: 일치하는 경우에만 실제 만료 삭제를 수행합니다 (stale entry 방어).
            # [2차]: 진짜 상한 물건이 맞다면 장부에서 지우고 쓰레기통에 버려달라고 부탁합니다.
            if current_expire_at is not None and current_expire_at == expire_at:
                # [1차]: 장부에서 키를 제거합니다.
                # [2차]: 장부에서 지웁니다.
                self._expire_map.remove(key)
                # [1차]: del_callback을 호출하여 저장소와 메모리에서 키를 완전 삭제합니다.
                # [2차]: "이 김밥 만료됐으니 매장에서 싹 치워주세요" 하고 폐기 처리합니다.
                del_callback(key, is_expired=True)

# [1차]: MiniRedis 코어 엔진 모듈을 정의합니다.
# [2차]: 손님의 주문을 접수하고 저장소, 캐시, 유통기한, 저울을 지휘하는 만능 총괄 매니저입니다.
"""
MiniRedis 코어 엔진 모듈
내장 dict, set, collections를 일절 사용하지 않고
직접 구현한 HashMap, DoublyLinkedList, MinHeap을 조화시켜 Redis 명령어를 처리합니다.
"""

# [1차]: 시간 측정을 위한 표준 time 모듈을 임포트합니다.
# [2차]: 중앙 관제탑 시계를 임포트합니다.
import time
# [1차]: 주 저장소로 사용할 자체 HashMap 클래스를 임포트합니다.

from src.hash_map import HashMap

from src.lru_cache import LRUManager
# [1차]: 메모리 한도 및 바이트 회계를 담당하는 MemoryManager 클래스를 임포트합니다.
# [2차]: 가방 무게를 달아보는 정밀 저울을 가져옵니다.
from src.memory_manager import MemoryManager
# [1차]: 만료 시각 및 최소 힙 정돈을 담당하는 TTLManager 클래스를 임포트합니다.
# [2차]: 유통기한 담당자를 데려옵니다.
from src.ttl_manager import TTLManager


# [1차]: Redis 호환 프로토콜과 메모리/TTL/LRU 정책을 통합 제어하는 MiniRedis 클래스를 선언합니다.
# [2차]: Mini Redis 매장의 총사령관 클래스입니다.
class MiniRedis:
    """
    Mini Redis 코어 엔진
    - Key-Value Storage: 자체 HashMap
    - LRU Tracking: LRUManager (DoublyLinkedList + HashMap)
    - TTL Management: TTLManager (MinHeap + HashMap)
    - Memory Management: MemoryManager
    """
    # [1차]: MiniRedis 인스턴스를 초기화하며 주 저장소, LRU, TTL, 메모리 매니저를 바인딩합니다.
    # [2차]: 매장 문을 열고 4명의 전문 직원(저장소, 책꽂이, 유통기한, 저울)을 채용하여 배치합니다.
    def __init__(self, time_provider=time.time):
        # [1차]: 시간 제공자 함수를 인스턴스 변수에 저장합니다.
        # [2차]: 매장 공식 표준시계를 설정합니다.
        self._time_provider = time_provider
        # [1차]: 키-값을 보관할 메인 HashMap을 인스턴스화합니다.
        # [2차]: 물건을 실제로 보관할 메인 수납장을 세웁니다.
        self._storage = HashMap()
        # [1차]: LRU 추적용 LRUManager를 인스턴스화합니다.
        # [2차]: LRU 책꽂이 순서판을 엽니다.
        self._lru = LRUManager()
        # [1차]: TTL 만료 관리용 TTLManager를 인스턴스화합니다.
        # [2차]: 유통기한 최소 힙 진열대를 준비합니다.
        self._ttl = TTLManager(time_provider=self._time_provider)
        # [1차]: 바이트 단위 메모리 관리를 위한 MemoryManager를 인스턴스화합니다.
        # [2차]: 전자저울을 켜둡니다.
        self._memory = MemoryManager()

    # [1차]: 저장소, LRU, TTL, 메모리 매니저에서 특정 키의 엔트리를 원자적으로 일괄 삭제합니다.
    # [2차]: 매장에서 물건을 뺄 때 4군데 장부에서 단 1개의 누락도 없이 깔끔하게 한 번에 지웁니다.
    def _delete_entry(self, key, is_eviction=False, is_expired=False):
        """
        내부 엔트리 완전 삭제 헬퍼 (저장소, LRU, TTL, 메모리에서 일괄 제거)
        """
        # [1차]: 메인 저장소 HashMap에서 키를 제거하고 기존 값을 가져옵니다.
        # [2차]: 메인 수납장에서 물건을 꺼냅니다.
        val = self._storage.remove(key)
        # [1차]: 기존 값이 존재하는 경우에만 나머지 구조에서도 동기화 제거를 수행합니다.
        # [2차]: 물건이 진짜 있었던 경우에만 다른 장부들도 정리합니다.
        if val is not None:
            # [1차]: LRU 관리자에서 해당 키를 O(1)으로 제거합니다.
            # [2차]: LRU 책꽂이 명부에서 지웁니다.
            self._lru.remove(key)
            # [1차]: TTL 관리자에서 해당 키의 만료 설정을 제거합니다.
            # [2차]: 유통기한 장부에서 지웁니다.
            self._ttl.remove_ttl(key)
            # [1차]: 메모리 관리자에서 해당 엔트리의 UTF-8 바이트 크기를 차감합니다.
            # [2차]: 저울에서 빠진 물건 무게를 뺍니다.
            self._memory.remove_entry_memory(key, val)
            # [1차]: 삭제 성공을 알리는 True를 반환합니다.
            # [2차]: 성공 보고를 올립니다.
            return True
        # [1차]: 삭제할 키가 없으면 False를 반환합니다.
        # [2차]: 없었음을 보고합니다.
        return False

    # [1차]: 키가 만료되었는지 사전에 검사하고 만료되었으면 즉각 삭제합니다.
    # [2차]: 손님이 찾기 전에 유통기한이 지났는지 먼저 쓱 확인하고 상했으면 그 자리에서 버립니다.
    def _check_and_expire(self, key):
        """
        키가 만료되었는지 확인하고, 만료되었으면 즉시 삭제합니다.
        삭제되었으면 True, 여전히 유효하거나 없으면 False 반환
        """
        # [1차]: 저장소에 키가 아예 없다면 만료 검사가 불필요하므로 False를 반환합니다.
        # [2차]: 매장에 아예 없는 물건이면 검사할 것도 없습니다.
        if not self._storage.contains(key):
            # [1차]: False 반환
            # [2차]: 없음 보고
            return False
        
        # [1차]: TTL 관리자에게 해당 키가 만료 시각을 지났는지 확인합니다.
        # [2차]: 유통기한 담당자에게 상했는지 물어봅니다.
        if self._ttl.is_expired(key):
            # [1차]: 만료되었으므로 내부 삭제 헬퍼를 호출하여 일괄 제거합니다.
            # [2차]: 상했으니 매장에서 즉시 폐기 처리합니다.
            self._delete_entry(key, is_expired=True)
            # [1차]: 만료 삭제되었음을 뜻하는 True를 반환합니다.
            # [2차]: 만료 폐기 완료 보고를 올립니다.
            return True
        # [1차]: 아직 신선하고 유효한 경우 False를 반환합니다.
        # [2차]: 안전함을 확인합니다.
        return False

    # [1차]: 인자 리스트를 받아 명령어를 파싱하고 적절한 서브 루틴을 라우팅하여 실행 결과를 반환합니다.
    # [2차]: 손님이 카운터에서 외친 명령어를 듣고 알맞은 담당 직원에게 일을 시킵니다.
    def execute_command(self, args):
        """
        파싱된 인자 리스트(예: ['SET', 'name', 'Alice'])를 받아 적절한 명령어를 실행하고
        Redis 표준 응답 문자열을 반환합니다.
        """
        # [1차]: 인자가 빈 리스트인 경우 빈 문자열을 반환합니다.
        # [2차]: 손님이 아무 말도 안 했으면 가만히 있습니다.
        if not args:
            # [1차]: 빈 문자열 반환
            # [2차]: 침묵
            return ""

        # [1차]: 명령어 이름을 대문자로 정규화합니다.
        # [2차]: 손님이 소문자로 말하든 대문자로 말하든 알아듣기 쉽게 대문자로 통일합니다.
        cmd = args[0].upper()

        # [1차]: 명령어 종류에 따라 분기하여 전용 처리 함수를 호출합니다.
        # [2차]: 외친 주문 메뉴에 따라 담당 요리사에게 접수합니다.
        if cmd == "SET":
            # [1차]: SET 명령어 실행
            # [2차]: 저장하기
            return self._cmd_set(args)
        elif cmd == "GET":
            # [1차]: GET 명령어 실행
            # [2차]: 꺼내보기
            return self._cmd_get(args)
        elif cmd == "DEL":
            # [1차]: DEL 명령어 실행
            # [2차]: 지우기
            return self._cmd_del(args)
        elif cmd == "EXISTS":
            # [1차]: EXISTS 명령어 실행
            # [2차]: 있는지 확인
            return self._cmd_exists(args)
        elif cmd == "DBSIZE":
            # [1차]: DBSIZE 명령어 실행
            # [2차]: 총 개수 세기
            return self._cmd_dbsize(args)
        elif cmd == "KEYS":
            # [1차]: KEYS 명령어 실행
            # [2차]: 전체 목록 보기
            return self._cmd_keys(args)
        elif cmd == "CONFIG":
            # [1차]: CONFIG 명령어 실행
            # [2차]: 설정 변경
            return self._cmd_config(args)
        elif cmd == "INFO":
            # [1차]: INFO 명령어 실행
            # [2차]: 정보 조회
            return self._cmd_info(args)
        elif cmd == "EXPIRE":
            # [1차]: EXPIRE 명령어 실행
            # [2차]: 유통기한 붙이기
            return self._cmd_expire(args)
        elif cmd == "TTL":
            # [1차]: TTL 명령어 실행
            # [2차]: 남은 시간 조회
            return self._cmd_ttl(args)
        else:
            # [1차]: 정의되지 않은 알 수 없는 명령어 에러를 Redis 표준 포맷으로 반환합니다.
            # [2차]: 메뉴판에 없는 메뉴를 시키면 그런 메뉴 없다고 정중히 거절합니다.
            return f"(error) ERR unknown command '{args[0]}'"

    # [1차]: SET key value 명령어를 처리합니다.
    # [2차]: 수납장에 새 물건을 꽂고 저울과 책꽂이 순서를 갱신합니다.
    def _cmd_set(self, args):
        """SET key value"""
        # [1차]: 인자 개수가 정확히 3개(SET, key, value)인지 검증합니다.
        # [2차]: 주문 인자 수가 안 맞으면 주문서를 반려합니다.
        if len(args) != 3:
            # [1차]: Redis 인자 개수 오류 반환
            # [2차]: 에러 통보
            return "(error) ERR wrong number of arguments for 'set' command"

        # [1차]: key와 value를 추출합니다.
        # [2차]: 이름표와 내용물을 손에 쥡니다.
        key, value = args[1], args[2]

        # [1차]: 단일 엔트리 자체가 maxmemory를 초과하는지 OOM을 사전 검사합니다.
        # [2차]: 이 물건 1개의 무게가 가방 전체 용량보다 더 무거운지 저울에 대봅니다.
        if self._memory.is_single_entry_oom(key, value):
            # [1차]: OOM 에러 메시지 반환
            # [2차]: 가방 터짐 에러 통보
            return "(error) OOM command not allowed when used_memory > 'maxmemory'"

        # [1차]: 실행 전 기존 키가 만료 상태인지 확인하고 지웁니다.
        # [2차]: 혹시 예전에 만료된 채 방치된 물건이 있다면 치웁니다.
        self._check_and_expire(key)

        # [1차]: 기존 키의 값이 존재하는지 조회합니다.
        # [2차]: 이미 꽂혀 있던 물건이 있는지 봅니다.
        existing_val = self._storage.get(key)
        # [1차]: 기존 값이 존재하는 경우 (덮어쓰기 분기)
        # [2차]: 덮어쓰기 상황입니다.
        if existing_val is not None:
            # [1차]: 덮어쓰기 시 기존 키에 걸려있던 TTL 설정을 규격에 따라 초기화(삭제)합니다.
            # [2차]: 새 물건으로 갈아 끼웠으니 이전 유통기한 딱지를 싹 떼어냅니다.
            self._ttl.remove_ttl(key)
            # [1차]: 값 차이만큼 메모리 사용량을 갱신합니다.
            # [2차]: 내용물 교체로 인한 무게 변화를 저울에 반영합니다.
            self._memory.update_entry_memory(key, existing_val, value)
            # [1차]: 주 저장소에 새 값을 put합니다.
            # [2차]: 수납장에 새 물건을 꽂습니다.
            self._storage.put(key, value)
        else:
            # [1차]: 신규 키 삽입 분기
            # [2차]: 완전히 새로운 물건을 처음 들여놓는 상황입니다.
            self._memory.add_entry_memory(key, value)
            # [1차]: 주 저장소에 키-값을 저장합니다.
            # [2차]: 수납장에 새 자리를 만들어 꽂습니다.
            self._storage.put(key, value)

        # [1차]: LRU 캐시 관리자에게 키 접근을 통보하여 맨 앞으로 승격시킵니다.
        # [2차]: 이 책을 책꽂이 맨 앞 1등석으로 옮깁니다.
        self._lru.access(key)

        # [1차]: 메모리 한도가 초과되었다면 한도 이하가 될 때까지 LRU 키들을 지속 방출합니다.
        # [2차]: 가방이 꽉 찼으면 한도 안으로 들어올 때까지 가장 오래 안 본 책부터 밖으로 던집니다.
        self._memory.evict_if_needed(self._lru, self._delete_entry)

        # [1차]: 성공 응답 'OK'를 반환합니다.
        # [2차]: 잘 저장되었다고 'OK' 도장을 찍어줍니다.
        return "OK"

    # [1차]: GET key 명령어를 처리합니다.
    # [2차]: 수납장에서 물건을 꺼내어 손님에게 보여줍니다.
    def _cmd_get(self, args):
        """GET key"""
        # [1차]: 인자 개수가 2개(GET, key)인지 검증합니다.
        # [2차]: 인자가 안 맞으면 거절합니다.
        if len(args) != 2:
            # [1차]: 인자 개수 오류 반환
            # [2차]: 에러 통보
            return "(error) ERR wrong number of arguments for 'get' command"

        # [1차]: 대상 key를 추출합니다.
        # [2차]: 찾으려는 이름표를 확인합니다.
        key = args[1]

        # [1차]: 키의 만료 여부를 검사하여 만료되었으면 삭제 후 (nil)을 반환합니다.
        # [2차]: 꺼내기 전에 유통기한을 보니 상했으면 바로 버리고 "(nil)"(없음)이라고 답합니다.
        if self._check_and_expire(key):
            # [1차]: (nil) 반환 (LRU 갱신 없음)
            # [2차]: 상해서 버렸으니 없음 통보
            return "(nil)"

        # [1차]: 저장소에서 값을 조회합니다.
        # [2차]: 수납장에서 물건을 꺼냅니다.
        val = self._storage.get(key)
        # [1차]: 값이 없으면 (nil)을 반환합니다.
        # [2차]: 원래 없던 물건이면 "(nil)"이라고 답합니다.
        if val is None:
            # [1차]: (nil) 반환
            # [2차]: 없음 통보
            return "(nil)"

        # [1차]: 조회 성공 시에만 LRU 순서를 최신으로 갱신합니다.
        # [2차]: 손님이 읽어본 책이니 책꽂이 맨 앞자리로 옮깁니다.
        self._lru.access(key)
        # [1차]: 따옴표로 감싼 Redis 스타일 문자열 포맷으로 값을 반환합니다.
        # [2차]: 예쁜 따옴표 상자에 담아 손님에게 건넵니다.
        return f'"{val}"'

    # [1차]: DEL key 명령어를 처리합니다.
    # [2차]: 매장에서 특정 물건을 영구 폐기합니다.
    def _cmd_del(self, args):
        """DEL key"""
        # [1차]: 인자 개수 검증
        # [2차]: 주문서 검증
        if len(args) != 2:
            # [1차]: 인자 오류
            # [2차]: 에러 통보
            return "(error) ERR wrong number of arguments for 'del' command"

        # [1차]: 대상 key 추출
        # [2차]: 버릴 이름표 확인
        key = args[1]

        # [1차]: 삭제 전 만료 확인 (만료 시 이미 삭제되어 (integer) 0 반환)
        # [2차]: 만료되어 이미 사라진 물건이면 지운 게 0개라고 알립니다.
        if self._check_and_expire(key):
            # [1차]: 0 반환
            # [2차]: 0개 삭제 보고
            return "(integer) 0"

        # [1차]: 내부 삭제 헬퍼를 호출하여 완전 삭제를 시도합니다.
        # [2차]: 4개 장부에서 싹 지웁니다.
        deleted = self._delete_entry(key)
        # [1차]: 삭제 성공 시 (integer) 1, 미존재 시 (integer) 0 반환
        # [2차]: 1개 지웠으면 1, 없었으면 0을 돌려줍니다.
        return "(integer) 1" if deleted else "(integer) 0"

    # [1차]: EXISTS key 명령어를 처리합니다.
    # [2차]: 매장에 이 물건이 현재 있는지 확인합니다.
    def _cmd_exists(self, args):
        """EXISTS key"""
        # [1차]: 인자 개수 검증
        # [2차]: 주문서 검증
        if len(args) != 2:
            # [1차]: 인자 오류
            # [2차]: 에러 통보
            return "(error) ERR wrong number of arguments for 'exists' command"

        # [1차]: key 추출
        # [2차]: 확인할 이름표
        key = args[1]
        # [1차]: 사전 만료 검사
        # [2차]: 유통기한 확인
        self._check_and_expire(key)

        # [1차]: 저장소에 키가 존재하는지 조회합니다.
        # [2차]: 수납장에 물건이 꽂혀 있는지 봅니다.
        exists = self._storage.contains(key)
        # [1차]: 존재 시 (integer) 1, 미존재 시 (integer) 0 반환
        # [2차]: 있으면 1, 없으면 0을 돌려줍니다.
        return "(integer) 1" if exists else "(integer) 0"

    # [1차]: DBSIZE 명령어를 처리합니다.
    # [2차]: 현재 매장에 진열된 총 유효 상품 개수를 셉니다.
    def _cmd_dbsize(self, args):
        """DBSIZE"""
        # [1차]: 인자 개수 검증
        # [2차]: 주문서 검증
        if len(args) != 1:
            # [1차]: 인자 오류
            # [2차]: 에러 통보
            return "(error) ERR wrong number of arguments for 'dbsize' command"

        # [1차]: 최소 힙을 통해 만료된 키들을 능동 정리합니다.
        # [2차]: 진열대 꼭대기를 훑어 상한 물건들을 싹 걷어냅니다.
        self._ttl.cleanup_expired(self._delete_entry)

        # [1차]: 전체 키를 순회하며 만료된 잔재가 있는지 재확인합니다.
        # [2차]: 혹시 남아있을지 모르는 상한 물건을 한 번 더 솎아냅니다.
        keys = self._storage.keys()
        for k in keys:
            self._check_and_expire(k)

        # [1차]: 현재 유효한 총 키 개수를 (integer) N 형식으로 반환합니다.
        # [2차]: 총 상품 숫자를 또박또박 셉니다.
        return f"(integer) {self._storage.size()}"

    # [1차]: KEYS 명령어를 처리하여 활성 키 목록을 배열 형식으로 출력합니다.
    # [2차]: 현재 매장에 남아있는 모든 물건의 이름표 목록을 출력합니다.
    def _cmd_keys(self, args):
        """KEYS"""
        # [1차]: 인자 개수 검증
        # [2차]: 주문서 검증
        if len(args) != 1:
            # [1차]: 인자 오류
            # [2차]: 에러 통보
            return "(error) ERR wrong number of arguments for 'keys' command"

        # [1차]: 만료된 항목을 능동 청소합니다.
        # [2차]: 상한 물건을 먼저 싹 정리합니다.
        self._ttl.cleanup_expired(self._delete_entry)
        # [1차]: 원시 키 목록을 가져옵니다.
        # [2차]: 수납장에 꽂힌 이름표들을 꺼냅니다.
        raw_keys = self._storage.keys()
        # [1차]: 유효 키들을 담을 리스트를 준비합니다.
        # [2차]: 진짜 신선한 이름표만 적을 명부를 만듭니다.
        valid_keys = []
        # [1차]: 각 키의 만료 여부를 검사하며 유효한 키만 수집합니다.
        # [2차]: 하나씩 냄새를 맡아보고 안전한 것만 명부에 올립니다.
        for k in raw_keys:
            if not self._check_and_expire(k):
                valid_keys.append(k)

        # [1차]: 유효한 키가 하나도 없으면 (empty array)를 반환합니다.
        # [2차]: 매장이 텅 비었으면 "(empty array)"라고 알립니다.
        if not valid_keys:
            # [1차]: 비어있음 응답
            # [2차]: 빈 배열 통보
            return "(empty array)"

        # [1차]: Redis 스타일 배열(1) "key" \n 2) "key") 포맷 문자열을 조립합니다.
        # [2차]: 번호 매겨진 깔끔한 목록으로 포장합니다.
        lines = []
        for i, k in enumerate(valid_keys, 1):
            lines.append(f'{i}) "{k}"')
        # [1차]: 개행 문자로 조립하여 반환합니다.
        # [2차]: 한 줄씩 출력합니다.
        return "\n".join(lines)

    # [1차]: CONFIG SET maxmemory bytes 명령어를 처리합니다.
    # [2차]: 매장 가방의 무게 제한 한도를 변경합니다.
    def _cmd_config(self, args):
        """CONFIG SET maxmemory bytes"""
        # [1차]: 인자 구성이 CONFIG SET maxmemory <bytes> 규격인지 검증합니다.
        # [2차]: 설정 문법이 올바른지 확인합니다.
        if len(args) != 4 or args[1].upper() != "SET" or args[2].lower() != "maxmemory":
            # [1차]: 문법 오류 반환
            # [2차]: 잘못된 설정 명령 거절
            return "(error) ERR unknown command or wrong arguments for 'config'"

        # [1차]: 바이트 인자를 정수로 파싱합니다.
        # [2차]: 숫자로 똑바로 적었는지 봅니다.
        try:
            # [1차]: 정수형 변환
            # [2차]: 정수 변환
            bytes_val = int(args[3])
            # [1차]: 0 미만의 음수는 허용되지 않습니다.
            # [2차]: 마이너스 무게는 말이 안 되니 거절합니다.
            if bytes_val < 0:
                return "(error) ERR value is not an integer or out of range"
        except ValueError:
            # [1차]: 정수 파싱 실패 오류 반환
            # [2차]: 숫자가 아니면 거절
            return "(error) ERR value is not an integer or out of range"

        # [1차]: 새 maxmemory 값을 메모리 매니저에 설정합니다.
        # [2차]: 저울의 최대 한도 눈금을 새 값으로 맞춥니다.
        self._memory.maxmemory = bytes_val
        # [1차]: 한도가 줄어들어 현재 사용량을 초과한 경우 즉시 LRU 방출을 수행합니다.
        # [2차]: 가방을 좁은 가방으로 바꿨으니 넘치는 책들을 즉각 밖으로 던집니다.
        self._memory.evict_if_needed(self._lru, self._delete_entry)
        # [1차]: 성공 응답 'OK' 반환
        # [2차]: 설정 완료 'OK'
        return "OK"

    # [1차]: INFO memory 명령어를 처리하여 메모리 통계를 출력합니다.
    # [2차]: 현재 저울 상태(사용 무게, 한도 무게, 버린 책 수) 영수증을 뽑아줍니다.
    def _cmd_info(self, args):
        """INFO memory"""
        # [1차]: 인자 규격 검증
        # [2차]: 주문서 검증
        if len(args) != 2 or args[1].lower() != "memory":
            # [1차]: 인자 오류
            # [2차]: 에러 통보
            return "(error) ERR wrong number of arguments for 'info' command"

        # [1차]: 메모리 매니저의 포맷팅 결과를 반환합니다.
        # [2차]: 통계 보고서를 전달합니다.
        return self._memory.get_info_memory()

    # [1차]: EXPIRE key seconds 명령어를 처리하여 만료 시간을 설정합니다.
    # [2차]: 특정 상품에 유통기한 딱지를 새로 붙입니다.
    def _cmd_expire(self, args):
        """EXPIRE key seconds"""
        # [1차]: 인자 개수 검증 (EXPIRE, key, seconds)
        # [2차]: 주문서 검증
        if len(args) != 3:
            # [1차]: 인자 오류
            # [2차]: 에러 통보
            return "(error) ERR wrong number of arguments for 'expire' command"

        # [1차]: 대상 key 추출
        # [2차]: 상품 이름표
        key = args[1]
        # [1차]: seconds 정수 파싱 검증
        # [2차]: 초 단위 숫자가 올바른지 확인
        try:
            seconds = int(args[2])
        except ValueError:
            # [1차]: 정수 오류 반환
            # [2차]: 숫자 에러 통보
            return "(error) ERR value is not an integer or out of range"

        # [1차]: 실행 전 사전 만료 검사
        # [2차]: 이미 유통기한 지난 상품인지 확인
        if self._check_and_expire(key):
            # [1차]: 없는 키로 간주되어 0 반환
            # [2차]: 0 반환
            return "(integer) 0"

        # [1차]: 저장소에 키가 없으면 (integer) 0 반환
        # [2차]: 매장에 없는 상품이면 딱지를 붙일 수 없으니 0을 반환합니다.
        if not self._storage.contains(key):
            # [1차]: 0 반환
            # [2차]: 실패 보고
            return "(integer) 0"

        # [1차]: seconds가 0 이하이면 즉시 만료 규칙에 따라 삭제합니다.
        # [2차]: 유통기한을 0초 이하로 정하면 즉시 상한 것이므로 즉각 쓰레기통에 버립니다.
        if seconds <= 0:
            # [1차]: 내부 삭제 수행
            # [2차]: 즉각 폐기
            self._delete_entry(key, is_expired=True)
            # [1차]: (integer) 1 반환
            # [2차]: 처리 완료 보고
            return "(integer) 1"

        # [1차]: TTL 관리자에 유효 초를 등록합니다.
        # [2차]: 유통기한 딱지를 찰떡같이 붙여둡니다.
        self._ttl.set_ttl(key, seconds)
        # [1차]: 성공 시 (integer) 1 반환
        # [2차]: 1 반환
        return "(integer) 1"

    # [1차]: TTL key 명령어를 처리하여 남은 만료 초를 반환합니다.
    # [2차]: 상품의 남은 유통기한을 조회합니다.
    def _cmd_ttl(self, args):
        """TTL key"""
        # [1차]: 인자 개수 검증
        # [2차]: 주문서 검증
        if len(args) != 2:
            # [1차]: 인자 오류
            # [2차]: 에러 통보
            return "(error) ERR wrong number of arguments for 'ttl' command"

        # [1차]: 대상 key 추출
        # [2차]: 상품 이름표
        key = args[1]
        # [1차]: 사전 만료 검사
        # [2차]: 유통기한 지남 여부 확인
        self._check_and_expire(key)

        # [1차]: 저장소에 키가 실제로 존재하는지 확인합니다.
        # [2차]: 매장에 물건이 있는지 봅니다.
        key_exists = self._storage.contains(key)
        # [1차]: TTL 관리자로부터 남은 초(-2, -1, N)를 계산받습니다.
        # [2차]: 유통기한 담당자에게 남은 초를 물어봅니다.
        ttl_val = self._ttl.get_ttl_seconds(key, key_exists)
        # [1차]: (integer) N 형식으로 반환합니다.
        # [2차]: 손님에게 결과를 알려줍니다.
        return f"(integer) {ttl_val}"

# [1차]: 비즈니스 로직을 수행할 MiniRedis 코어 엔진을 임포트합니다.

from src.mini_redis import MiniRedis


# [1차]: 사용자 입력 문자열을 공백 및 따옴표 규칙에 따라 토큰 리스트로 분할합니다.
# [2차]: 손님의 긴 말 한마디를 하나하나의 단어 주문서로 예쁘게 잘라냅니다.
def parse_line(line):
    """
    명령줄 문자열을 공백 및 따옴표 규칙에 맞추어 분할합니다.
    shlex를 사용하여 큰따옴표/작은따옴표로 감싸진 문자열을 원자적으로 파싱합니다.
    """
    # [1차]: 정상적인 shlex.split 파싱을 시도합니다.
    # [2차]: 따옴표 문법을 지키며 쪼개봅니다.
    try:
        # [1차]: shlex.split(line)으로 분할된 인자 리스트를 반환합니다.
        # [2차]: 쪼갠 단어 목록을 건넵니다.
        return shlex.split(line)
    # [1차]: 따옴표가 닫히지 않은 경우 ValueError 예외를 잡습니다.
    # [2차]: 따옴표를 깜빡 잊고 안 닫았을 때 당황하지 않고 대처합니다.
    except ValueError:
        # [1차]: 폴백으로 단순 공백 분할을 수행합니다.
        # [2차]: 그냥 단순 띄어쓰기 기준으로 쪼개서 반환합니다.
        return line.strip().split()


# [1차]: 터미널에서 사용자 입력을 반복 대기하고 처리하는 메인 REPL 루프 함수를 정의합니다.
# [2차]: 매장 문을 열고 손님이 나갈 때까지 계속 주문을 받는 카운터 루프입니다.
def run_cli():
    """
    Mini Redis REPL 루프
    """
    # [1차]: MiniRedis 코어 엔진 인스턴스를 생성합니다.
    # [2차]: 오늘 영업을 시작할 총괄 매니저를 배치합니다.
    redis = MiniRedis()
    # [1차]: 환영 안내 메시지를 출력합니다.
    # [2차]: "어서오세요 Mini Redis 매장입니다!" 하고 인사합니다.
    print("Welcome to Mini Redis CLI! Type 'exit' or 'quit' to close.")
    # [1차]: 의존성 없는 순수 구현체임을 안내합니다.
    # [2차]: "외부 재료 없이 100% 수제로 만든 매장입니다" 하고 자랑합니다.
    print("Zero-dependency from-scratch key-value store.")

    # [1차]: 종료 명령이 입력될 때까지 무한 반복하는 REPL 루프를 시작합니다.
    # [2차]: 손님이 집에 갈 때까지 계속 주문을 기다립니다.
    while True:
        # [1차]: 예외 처리를 위한 try 블록을 시작합니다.
        # [2차]: 만약 손님이 Ctrl+C를 눌러도 안전하게 처리할 준비를 합니다.
        try:
            # [1차]: 프롬프트 'mini-redis> '를 출력하고 사용자 입력을 받아 양 끝 공백을 제거합니다.
            # [2차]: "주문하시겠어요?" 마이크를 건네고 손님의 말을 듣습니다.
            line = input("mini-redis> ").strip()
            # [1차]: 빈 입력인 경우 다음 입력으로 넘어갑니다.
            # [2차]: 아무 말도 안 하고 엔터만 쳤으면 다시 기다립니다.
            if not line:
                # [1차]: 루프 계속
                # [2차]: 다음 주문 대기
                continue

            # [1차]: 입력이 exit 또는 quit이면 루프를 종료합니다.
            # [2차]: "나갈게요"라고 하면 영업을 종료합니다.
            if line.lower() in ("exit", "quit"):
                # [1차]: 작별 인사 출력
                # [2차]: "안녕히 가세요!" 인사
                print("Bye!")
                # [1차]: 루프 탈출
                # [2차]: 퇴장
                break

            # [1차]: 입력 라인을 인자 리스트로 파싱합니다.
            # [2차]: 손님의 말을 주문서 조각들로 쪼갭니다.
            args = parse_line(line)
            # [1차]: 파싱 결과가 비어있으면 다음으로 건너뜁니다.
            # [2차]: 주문 내용이 없으면 넘어갑니다.
            if not args:
                # [1차]: 루프 계속
                # [2차]: 다음 주문 대기
                continue

            # [1차]: MiniRedis 코어 엔진에 인자를 넘겨 실행하고 응답을 받습니다.
            # [2차]: 주방 매니저에게 주문서를 넘기고 완성된 요리(응답)를 받아옵니다.
            response = redis.execute_command(args)
            # [1차]: 응답이 존재하는 경우 터미널에 출력합니다.
            # [2차]: 손님에게 결과를 건네며 영수증을 보여줍니다.
            if response:
                # [1차]: 응답 문자열 출력
                # [2차]: 화면에 출력
                print(response)

        # [1차]: 사용자가 Ctrl+C 또는 Ctrl+D(EOF)로 인터럽트를 발생시킨 경우를 포착합니다.
        # [2차]: 손님이 급하게 창문을 닫거나 나갔을 때 깔끔하게 마무리합니다.
        except (KeyboardInterrupt, EOFError):
            # [1차]: 개행 및 작별 메시지 출력
            # [2차]: "안녕히 가세요!" 인사
            print("\nBye!")
            # [1차]: 루프 탈출
            # [2차]: 종료
            break
        # [1차]: 그 외 예기치 않은 시스템 예외를 포착합니다.
        # [2차]: 뜻밖의 사고가 발생해도 프로그램이 갑자기 튕기지 않게 방어합니다.
        except Exception as e:
            # [1차]: 에러 메시지를 포맷팅하여 출력합니다.
            # [2차]: 어떤 문제가 생겼는지 친절히 알려줍니다.
            print(f"(error) ERR unexpected system error: {e}")


# [1차]: 이 파일이 메인 모듈로 직접 실행되었을 때 run_cli를 호출합니다.
# [2차]: 이 파일을 직접 더블 클릭하면 카운터를 즉시 엽니다.
if __name__ == "__main__":
    # [1차]: CLI 루프 실행
    # [2차]: 카운터 개장
    run_cli()

    run_cli()
